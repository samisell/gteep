    96→    96→function AnimatedCounter({ end, suffix = '', duration = 2000 }: { end: number; suffix?: string; duration?: number }) {
    97→    97→  const [count, setCount] = useState(0);
    98→    98→  const ref = useRef<HTMLSpanElement>(null);
    99→    99→  const isInView = useInView(ref, { once: true });
   100→   100→  const hasStarted = useRef(false);
   101→   101→
   102→   102→  useEffect(() => {
   103→   103→    if (!isInView || hasStarted.current) return;
   104→   104→    hasStarted.current = true;
   105→   105→
   106→   106→    const startTime = performance.now();
   107→   107→    const animate = (currentTime: number) => {
   108→   108→      const elapsed = currentTime - startTime;
   109→   109→      const progress = Math.min(elapsed / duration, 1);
   110→   110→      const eased = 1 - Math.pow(1 - progress, 3);
   111→   111→      setCount(Math.floor(eased * end));
   112→   112→      if (progress < 1) {
   113→   113→        requestAnimationFrame(animate);
   114→   114→      }
   115→   115→    };
   116→   116→    requestAnimationFrame(animate);
   117→   117→  }, [isInView, end, duration]);
   118→   118→
   119→   119→  return (
   120→   120→    <span ref={ref} className="tabular-nums">
   121→   121→      {count}{suffix}
   122→   122→    </span>
   123→   123→  );
   124→   124→}
