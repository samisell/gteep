   136→// Download Form Dialog Component
   137→// =============================================================================
   138→
   139→function DownloadFormDialog({ fileUrl, fileName }: { fileUrl: string; fileName: string }) {
   140→  const [email, setEmail] = useState('');
   141→  const [name, setName] = useState('');
   142→  const [isSubmitting, setIsSubmitting] = useState(false);
   143→  const [isSubmitted, setIsSubmitted] = useState(false);
   144→  const [isOpen, setIsOpen] = useState(false);
   145→
   146→  const handleSubmit = async (e: React.FormEvent) => {
   147→    e.preventDefault();
   148→    if (!email.trim()) return;
   149→
   150→    setIsSubmitting(true);
   151→    try {
   152→      const res = await fetch('/api/download/request', {
   153→        method: 'POST',
   154→        headers: { 'Content-Type': 'application/json' },
   155→        body: JSON.stringify({ email, name, fileUrl, fileName }),
   156→      });
   157→      if (res.ok) {
   158→        setIsSubmitted(true);
   159→        // Also start the download directly
   160→        window.open(fileUrl, '_blank');
   161→      }
   162→    } catch {
   163→      // Still allow download even if API fails
   164→      window.open(fileUrl, '_blank');
   165→      setIsSubmitted(true);
   166→    } finally {
   167→      setIsSubmitting(false);
   168→    }
   169→  };
   170→
   171→  const handleOpenChange = (open: boolean) => {
   172→    setIsOpen(open);
   173→    if (!open) {
   174→      // Reset after a short delay so the user doesn't see the form reset
   175→      setTimeout(() => {
   176→        setIsSubmitted(false);
   177→        setEmail('');
   178→        setName('');
   179→      }, 300);
   180→    }
   181→  };
   182→
   183→  return (
   184→    <Dialog open={isOpen} onOpenChange={handleOpenChange}>
   185→      <DialogTrigger asChild>
   186→        <Button
   187→          size="sm"
   188→          className="bg-emerald-700 hover:bg-emerald-800 text-white rounded-lg"
   189→        >
   190→          <Download className="w-4 h-4 mr-2" />
   191→          Download Slides
   192→        </Button>
   193→      </DialogTrigger>
   194→      <DialogContent className="sm:max-w-md">
   195→        <DialogHeader>
   196→          <DialogTitle className="flex items-center gap-2">
   197→            <FileText className="w-5 h-5 text-emerald-600" />
   198→            Download Presentation
   199→          </DialogTitle>
   200→          <DialogDescription>
   201→            {isSubmitted
   202→              ? 'Your download has started!'
   203→              : 'Enter your details to download the presentation slides.'}
   204→          </DialogDescription>
   205→        </DialogHeader>
   206→        {isSubmitted ? (
   207→          <motion.div
   208→            initial={{ opacity: 0, scale: 0.9 }}
   209→            animate={{ opacity: 1, scale: 1 }}
   210→            className="py-8 text-center"
   211→          >
   212→            <CheckCircle2 className="w-16 h-16 text-emerald-500 mx-auto mb-4" />
   213→            <h3 className="text-lg font-semibold text-foreground mb-2">
   214→              Download Started!
   215→            </h3>
   216→            <p className="text-sm text-muted-foreground mb-4">
   217→              Thank you, {name || 'friend'}! Your download should begin automatically.
   218→              We&apos;ve also sent a confirmation to <strong>{email}</strong>.
   219→            </p>
   220→            <Button
   221→              variant="outline"
   222→              onClick={() => handleOpenChange(false)}
   223→              className="rounded-lg"
   224→            >
   225→              Close
   226→            </Button>
   227→          </motion.div>
   228→        ) : (
   229→          <form onSubmit={handleSubmit} className="space-y-4 pt-2">
   230→            <div className="space-y-2">
   231→              <label htmlFor="download-name" className="text-sm font-medium">
   232→                Full Name
   233→              </label>
   234→              <Input
   235→                id="download-name"
   236→                type="text"
   237→                placeholder="Your full name"
   238→                value={name}
   239→                onChange={(e) => setName(e.target.value)}
   240→                className="rounded-lg"
   241→              />
   242→            </div>
   243→            <div className="space-y-2">
   244→              <label htmlFor="download-email" className="text-sm font-medium">
   245→                Email Address <span className="text-red-500">*</span>
   246→              </label>
   247→              <Input
   248→                id="download-email"
   249→                type="email"
   250→                placeholder="your@email.com"
   251→                value={email}
   252→                onChange={(e) => setEmail(e.target.value)}
   253→                required
   254→                className="rounded-lg"
   255→              />
   256→            </div>
   257→            <p className="text-xs text-muted-foreground">
   258→              By downloading, you agree to receive occasional policy updates from GTEEP.
   259→              You can unsubscribe at any time.
   260→            </p>
   261→            <Button
   262→              type="submit"
   263→              disabled={isSubmitting || !email.trim()}
   264→              className="w-full bg-emerald-700 hover:bg-emerald-800 text-white rounded-lg"
   265→            >
   266→              {isSubmitting ? (
   267→                <>
   268→                  <Loader2 className="w-4 h-4 mr-2 animate-spin" />
   269→                  Processing...
   270→                </>
   271→              ) : (
   272→                <>
   273→                  <Download className="w-4 h-4 mr-2" />
   274→                  Download Now
   275→                </>
   276→              )}
   277→            </Button>
   278→          </form>
   279→        )}
   280→      </DialogContent>
   281→    </Dialog>
   282→  );
   283→}
