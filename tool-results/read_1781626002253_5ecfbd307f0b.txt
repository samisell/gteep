    68→// -----------------------------------------------------------------------------
    69→// Helper: Extract nodes from a GraphQL response
    70→// -----------------------------------------------------------------------------
    71→
    72→function extractNodes<T>(
    73→  response: { data?: any; errors?: any[] },
    74→  path: string
    75→): T[] {
    76→  if (response.errors || !response.data) {
    77→    return [];
    78→  }
    79→
    80→  const parts = path.split('.');
    81→  let result: any = response.data;
    82→  for (const part of parts) {
    83→    if (result?.[part] === undefined) return [];
    84→    result = result[part];
    85→  }
    86→
    87→  if (result?.nodes && Array.isArray(result.nodes) && result.nodes.length > 0) {
    88→    return result.nodes as T[];
    89→  }
    90→
    91→  return [];
    92→}
    93→
    94→function extractSingleNode<T>(
    95→  response: { data?: any; errors?: any[] },
    96→  key: string
    97→): T | null {
    98→  if (response.errors || !response.data) {
    99→    return null;
   100→  }
   101→
   102→  const node = response.data[key];
   103→  if (!node) return null;
   104→
   105→  return node as T;
   106→}
   107→
   108→function extractPageInfo(
   109→  response: { data?: any; errors?: any[] },
   110→  path: string
   111→): WPPageInfo | null {
   112→  if (response.errors || !response.data) return null;
   113→
   114→  const parts = path.split('.');
   115→  let result: any = response.data;
   116→  for (const part of parts) {
   117→    if (result?.[part] === undefined) return null;
   118→    result = result[part];
   119→  }
   120→
   121→  return result?.pageInfo ?? null;
   122→}

   250→// -----------------------------------------------------------------------------
   251→// Admin Email (fetches ACF adminemail field from WP)
   252→// The adminemail ACF field group is on the Post type, with a single field:
   253→//   adminEmailForNotification — the email that receives contact/newsletter notifications
   254→// -----------------------------------------------------------------------------
   255→
   256→const DEFAULT_ADMIN_EMAIL = 'info@gteep.gileadtrust.com';
   257→
   258→// Cache admin email for 5 minutes to avoid repeated WP queries
   259→let adminEmailCache: { email: string; timestamp: number } | null = null;
   260→const ADMIN_EMAIL_CACHE_TTL = 5 * 60 * 1000; // 5 minutes
   261→
   262→export async function getAdminEmail(): Promise<string> {
   263→  // Return cached value if still valid
   264→  if (adminEmailCache && Date.now() - adminEmailCache.timestamp < ADMIN_EMAIL_CACHE_TTL) {
   265→    return adminEmailCache.email;
   266→  }
   267→
   268→  try {
   269→    const response = await fetchGraphQL<any>(GET_ADMIN_EMAIL);
   270→
   271→    if (!response.errors && response.data?.posts?.nodes?.length) {
   272→      const email = response.data.posts.nodes[0]?.adminemail?.adminEmailForNotification;
   273→      if (email && /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
   274→        adminEmailCache = { email, timestamp: Date.now() };
   275→        return email;
   276→      }
   277→    }
   278→  } catch {
   279→    // Fall through to default
   280→  }
   281→
   282→  // Fall back to env var or default
   283→  const fallback = process.env.CONTACT_RECEIVER_EMAIL || process.env.SITE_EMAIL || DEFAULT_ADMIN_EMAIL;
   284→  return fallback;
   285→}

   337→/**
   338→ * Fetch the site logo URL from WordPress.
   339→ */
   340→export async function getSiteLogo(): Promise<string | null> {
   341→  try {
   342→    const response = await fetchGraphQL<any>(`
   343→      query GetSiteLogo {
   344→        mediaItems(first: 10, where: { search: "cropped-refined_logo" }) {
   345→          nodes {
   346→            id
   347→            sourceUrl
   348→            altText
   349→            title
   350→          }
   351→        }
   352→      }
   353→    `);
   354→
   355→    if (response.errors || !response.data?.mediaItems?.nodes?.length) {
   356→      const fallbackResponse = await fetchGraphQL<any>(`
   357→        query GetSiteIcon {
   358→          mediaItem(id: "8", idType: DATABASE_ID) {
   359→            sourceUrl
   360→            altText
   361→          }
   362→        }
   363→      `);
   364→
   365→      if (fallbackResponse.errors || !fallbackResponse.data?.mediaItem?.sourceUrl) {
   366→        return null;
   367→      }
   368→
   369→      return fallbackResponse.data.mediaItem.sourceUrl;
   370→    }
   371→
   372→    const logo = response.data.mediaItems.nodes.find(
   373→      (node: any) => node.title?.includes('cropped-refined_logo') || node.sourceUrl?.includes('cropped-refined_logo')
   374→    );
   375→
   376→    return logo?.sourceUrl || response.data.mediaItems.nodes[0]?.sourceUrl || null;
   377→  } catch {
   378→    return null;
   379→  }
   380→}

   382→export async function getSiteSettings(): Promise<WPSiteSettings> {
   383→  const response = await fetchGraphQL<WPSiteSettingsData>(GET_SITE_SETTINGS);
   384→
   385→  if (response.errors || !response.data) {
   386→    return defaultSiteSettings;
   387→  }
   388→
   389→  const { generalSettings } = response.data;
   390→
   391→  if (!generalSettings) {
   392→    return defaultSiteSettings;
   393→  }
   394→
   395→  // Try to fetch the site logo from WP
   396→  const logoUrl = await getSiteLogo();
   397→
   398→  // Determine the site description
   399→  const wpDescription = generalSettings.description;
   400→  const isDefaultWpTagline = !wpDescription || wpDescription === 'Just another WordPress site';
   401→  const siteDescription = isDefaultWpTagline ? defaultSiteSettings.siteDescription : wpDescription;
   402→
   403→  return {
   404→    siteTitle: generalSettings.title || defaultSiteSettings.siteTitle,
   405→    siteDescription,
   406→    siteUrl: generalSettings.url || defaultSiteSettings.siteUrl,
   407→    siteLogo: logoUrl
   408→      ? {
   409→          sourceUrl: logoUrl,
   410→          altText: 'GTEEP Logo',
   411→          mediaItemId: 8,
   412→          width: 512,
   413→          height: 512,
   414→        }
   415→      : null,
   416→    favicon: logoUrl
   417→      ? {
   418→          sourceUrl: logoUrl,
   419→          altText: 'GTEEP Favicon',
   420→          mediaItemId: 8,
   421→          width: 512,
   422→          height: 512,
   423→        }
   424→      : null,
   425→    acfOptions: defaultSiteSettings.acfOptions,
   426→  };
   427→}

   501→/**
   502→ * Valid team member categories for ACF teamCategory select field
   503→ */
   504→const VALID_TEAM_CATEGORIES = ['executive', 'director', 'advisory-board', 'board-of-trustees'] as const;
   505→
   506→/**
   507→ * Get team members data from WordPress.
   508→ *
   509→ * Strategy: Try ACF first, then fall back to category-based approach.
   510→ *
   511→ * 1. ACF approach (preferred): Fetches posts with ACF "teamMember" field group.
   512→ *    - Requires ACF field group "Team Member" with GraphQL type "TeamMember"
   513→ *    - Uses a "team" category to scope the query
   514→ *    - Each post has ACF fields: teamName, teamRole, teamCategory, teamBio, teamImage
   515→ *
   516→ * 2. Category-based fallback (legacy): Fetches posts from 4 separate categories.
   517→ *    - Post Title = Person's Name
   518→ *    - First Tag = Role/Position
   519→ *    - Post Content/Excerpt = Bio
   520→ *    - Featured Image = Profile picture
   521→ *    - Category = Which group they belong to
   522→ */
   523→export async function getTeamMembers(): Promise<GTEEPTeamMember[]> {
   524→  // --- Try ACF approach first ---
   525→  try {
   526→    const acfResponse = await fetchGraphQL<any>(GET_TEAM_MEMBERS_ACF);
   527→
   528→    if (!acfResponse.errors && acfResponse.data?.posts?.nodes?.length) {
   529→      const posts = acfResponse.data.posts.nodes;
   530→
   531→      // Check if any posts actually have ACF teamMember data
   532→      const hasAcfData = posts.some(
   533→        (p: any) => p.teamMember && (p.teamMember.teamName || p.teamMember.teamRole)
   534→      );
   535→
   536→      if (hasAcfData) {
   537→        const teamMembers: GTEEPTeamMember[] = posts
   538→          .filter((p: any) => p.teamMember?.teamName || p.title)
   539→          .map((p: any) => {
   540→            const acf = p.teamMember || {};
   541→            const category = VALID_TEAM_CATEGORIES.includes(acf.teamCategory)
   542→              ? acf.teamCategory
   543→              : 'director'; // default fallback
   544→
   545→            return {
   546→              id: p.id,
   547→              name: acf.teamName || p.title || '',
   548→              role: acf.teamRole || (p.tags?.nodes?.[0]?.name || category.charAt(0).toUpperCase() + category.slice(1).replace(/-/g, ' ')),
   549→              category,
   550→              bio: acf.teamBio || (p.content
   551→                ? p.content.replace(/<[^>]*>/g, '').trim()
   552→                : p.excerpt?.replace(/<[^>]*>/g, '').trim() || ''),
   553→              image: acf.teamImage?.sourceUrl || p.featuredImage?.node?.sourceUrl || '',
   554→            };
   555→          });
   556→
   557→        return teamMembers;
   558→      }
   559→    }
   560→  } catch {
   561→    // ACF query failed — likely not set up yet. Fall through to category-based.
   562→  }
   563→
   564→  // --- Category-based fallback (original approach) ---
   565→  try {
   566→    const response = await fetchGraphQL<any>(GET_TEAM_MEMBERS);
   567→
   568→    if (response.errors || !response.data) {
   569→      return [];
   570→    }
   571→
   572→    const mapPostToMember = (
   573→      post: any,
   574→      category: GTEEPTeamMember['category']
   575→    ): GTEEPTeamMember => ({
   576→      id: post.id,
   577→      name: post.title || '',
   578→      role: post.tags?.nodes?.[0]?.name || category.charAt(0).toUpperCase() + category.slice(1).replace(/-/g, ' '),
   579→      category,
   580→      bio: post.content
   581→        ? post.content.replace(/<[^>]*>/g, '').trim()
   582→        : post.excerpt?.replace(/<[^>]*>/g, '').trim() || '',
   583→      image: post.featuredImage?.node?.sourceUrl || '',
   584→    });
   585→
   586→    const { executives, directors, advisoryBoard, boardOfTrustees } = response.data;
   587→
   588→    const teamMembers: GTEEPTeamMember[] = [
   589→      ...(executives?.nodes || []).map((p: any) => mapPostToMember(p, 'executive')),
   590→      ...(directors?.nodes || []).map((p: any) => mapPostToMember(p, 'director')),
   591→      ...(advisoryBoard?.nodes || []).map((p: any) => mapPostToMember(p, 'advisory-board')),
   592→      ...(boardOfTrustees?.nodes || []).map((p: any) => mapPostToMember(p, 'board-of-trustees')),
   593→    ];
   594→
   595→    return teamMembers;
   596→  } catch {
   597→    return [];
   598→  }
   599→}

   638→
   639→  // Categories used for team members — these should NOT appear in blog posts
   640→  const teamCategorySlugs = ['executive', 'director', 'advisory-board', 'board-of-trustees', 'team'];
   641→
   642→  // Filter out team member posts and the default "Hello world!" WP post
   643→  const blogPosts = posts.filter((post) => {
   644→    if (post.slug === 'hello-world') return false;
   645→
   646→    const postCategories = (post.categories as any)?.nodes || [];
   647→    const hasTeamCategory = postCategories.some((cat: any) =>
   648→      teamCategorySlugs.includes(cat.slug)
   649→    );
   650→    if (hasTeamCategory) return false;
   651→
   652→    return true;
   653→  });
   654→
   655→  if (blogPosts.length > 0) {
   656→    return blogPosts.map((post): import('@/types').GTEEPBlogPost => ({
   657→      id: post.id,
   658→      title: post.title,
   659→      slug: post.slug,
   660→      excerpt: post.excerpt || '',
   661→      content: post.content || '',
   662→      date: post.date,
   663→      author: (post.author as any)?.node?.name || 'GTEEP Team',
   664→      image: post.featuredImage?.node?.sourceUrl || undefined,
   665→      categories: (post.categories as any)?.nodes?.map((c: any) => c.name) || [],
   666→    }));
   667→  }
   668→
   669→  return [];
   670→}
