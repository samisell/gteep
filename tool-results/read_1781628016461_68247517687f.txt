    70→// -----------------------------------------------------------------------------
    71→// Helper: Extract nodes from a GraphQL response
    72→// -----------------------------------------------------------------------------
    73→
    74→function extractNodes<T>(
    75→  response: { data?: any; errors?: any[] },
    76→  path: string
    77→): T[] {
    78→  if (response.errors || !response.data) {
    79→    return [];
    80→  }
    81→
    82→  const parts = path.split('.');
    83→  let result: any = response.data;
    84→  for (const part of parts) {
    85→    if (result?.[part] === undefined) return [];
    86→    result = result[part];
    87→  }
    88→
    89→  if (result?.nodes && Array.isArray(result.nodes) && result.nodes.length > 0) {
    90→    return result.nodes as T[];
    91→  }
    92→
    93→  return [];
    94→}
    95→
    96→function extractSingleNode<T>(
    97→  response: { data?: any; errors?: any[] },
    98→  key: string
    99→): T | null {
   100→  if (response.errors || !response.data) {
   101→    return null;
   102→  }
   103→
   104→  const node = response.data[key];
   105→  if (!node) return null;
   106→
   107→  return node as T;
   108→}
   109→
   110→function extractPageInfo(
   111→  response: { data?: any; errors?: any[] },
   112→  path: string
   113→): WPPageInfo | null {
   114→  if (response.errors || !response.data) return null;
   115→
   116→  const parts = path.split('.');
   117→  let result: any = response.data;
   118→  for (const part of parts) {
   119→    if (result?.[part] === undefined) return null;
   120→    result = result[part];
   121→  }
   122→
   123→  return result?.pageInfo ?? null;
   124→}

   252→// -----------------------------------------------------------------------------
   253→// Admin Email (fetches ACF adminemail field from WP)
   254→// The adminemail ACF field group is on the Post type, with a single field:
   255→//   adminEmailForNotification — the email that receives contact/newsletter notifications
   256→// -----------------------------------------------------------------------------
   257→
   258→const DEFAULT_ADMIN_EMAIL = 'info@gteep.gileadtrust.com';
   259→
   260→// Cache admin email for 5 minutes to avoid repeated WP queries
   261→let adminEmailCache: { email: string; timestamp: number } | null = null;
   262→const ADMIN_EMAIL_CACHE_TTL = 5 * 60 * 1000; // 5 minutes
   263→
   264→export async function getAdminEmail(): Promise<string> {
   265→  // Return cached value if still valid
   266→  if (adminEmailCache && Date.now() - adminEmailCache.timestamp < ADMIN_EMAIL_CACHE_TTL) {
   267→    return adminEmailCache.email;
   268→  }
   269→
   270→  try {
   271→    const response = await fetchGraphQL<any>(GET_ADMIN_EMAIL);
   272→
   273→    if (!response.errors && response.data?.posts?.nodes?.length) {
   274→      const email = response.data.posts.nodes[0]?.adminemail?.adminEmailForNotification;
   275→      if (email && /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
   276→        adminEmailCache = { email, timestamp: Date.now() };
   277→        return email;
   278→      }
   279→    }
   280→  } catch {
   281→    // Fall through to default
   282→  }
   283→
   284→  // Fall back to env var or default
   285→  const fallback = process.env.CONTACT_RECEIVER_EMAIL || process.env.SITE_EMAIL || DEFAULT_ADMIN_EMAIL;
   286→  return fallback;
   287→}

   339→/**
   340→ * Fetch the site logo URL from WordPress.
   341→ */
   342→export async function getSiteLogo(): Promise<string | null> {
   343→  try {
   344→    const response = await fetchGraphQL<any>(`
   345→      query GetSiteLogo {
   346→        mediaItems(first: 10, where: { search: "cropped-refined_logo" }) {
   347→          nodes {
   348→            id
   349→            sourceUrl
   350→            altText
   351→            title
   352→          }
   353→        }
   354→      }
   355→    `);
   356→
   357→    if (response.errors || !response.data?.mediaItems?.nodes?.length) {
   358→      const fallbackResponse = await fetchGraphQL<any>(`
   359→        query GetSiteIcon {
   360→          mediaItem(id: "8", idType: DATABASE_ID) {
   361→            sourceUrl
   362→            altText
   363→          }
   364→        }
   365→      `);
   366→
   367→      if (fallbackResponse.errors || !fallbackResponse.data?.mediaItem?.sourceUrl) {
   368→        return null;
   369→      }
   370→
   371→      return fallbackResponse.data.mediaItem.sourceUrl;
   372→    }
   373→
   374→    const logo = response.data.mediaItems.nodes.find(
   375→      (node: any) => node.title?.includes('cropped-refined_logo') || node.sourceUrl?.includes('cropped-refined_logo')
   376→    );
   377→
   378→    return logo?.sourceUrl || response.data.mediaItems.nodes[0]?.sourceUrl || null;
   379→  } catch {
   380→    return null;
   381→  }
   382→}

   384→export async function getSiteSettings(): Promise<WPSiteSettings> {
   385→  const response = await fetchGraphQL<WPSiteSettingsData>(GET_SITE_SETTINGS);
   386→
   387→  if (response.errors || !response.data) {
   388→    return defaultSiteSettings;
   389→  }
   390→
   391→  const { generalSettings } = response.data;
   392→
   393→  if (!generalSettings) {
   394→    return defaultSiteSettings;
   395→  }
   396→
   397→  // Try to fetch the site logo from WP
   398→  const logoUrl = await getSiteLogo();
   399→
   400→  // Determine the site description
   401→  const wpDescription = generalSettings.description;
   402→  const isDefaultWpTagline = !wpDescription || wpDescription === 'Just another WordPress site';
   403→  const siteDescription = isDefaultWpTagline ? defaultSiteSettings.siteDescription : wpDescription;
   404→
   405→  return {
   406→    siteTitle: generalSettings.title || defaultSiteSettings.siteTitle,
   407→    siteDescription,
   408→    siteUrl: generalSettings.url || defaultSiteSettings.siteUrl,
   409→    siteLogo: logoUrl

   411→          sourceUrl: logoUrl,
   412→          altText: 'GTEEP Logo',
   413→          mediaItemId: 8,
   414→          width: 512,

   450→/**
   451→ * Get activities (What We Do) data from WordPress.
   452→ * Fetches the "what-we-do" page and its child pages from WP.
   453→ * Each child page becomes an activity section.
   454→ * Nested children (e.g., Policy Firechat under Policy Engagement) are included.
   455→ * New pages added under "What We Do" in WP will appear automatically.
   456→ */

   459→    const response = await fetchGraphQL<any>(GET_WHAT_WE_DO_PAGE);
   460→
   461→    if (response.errors || !response.data?.page) {
   462→      return [];
   463→    }
   464→
   465→    const page = response.data.page;
   466→    const childNodes = page.children?.nodes || [];
   467→
   468→    if (childNodes.length === 0) {
   469→      return [];
   470→    }
   471→
   472→    // Map of known icons for specific activities (by slug)
   473→    const iconMap: Record<string, string> = {
   474→      'policy-research': 'FileSearch',
   475→      'policy-engagement': 'Users',
   476→      'citizen-enlightenment': 'Lightbulb',
   477→      'data-speaks': 'BarChart3',
   478→      'youth-mentoring': 'GraduationCap',
   479→      'womens-economic-livelihood': 'Heart',
   480→    };
   481→
   482→    return childNodes.map((child: any, index: number) => {
   483→      // Parse nested children (e.g., Policy Firechat under Policy Engagement)
   484→      const nestedChildren: import('@/types').GTEEPActivityChild[] = (child.children?.nodes || []).map(
   485→        (nested: any) => ({
   486→          id: nested.id,
   487→          databaseId: nested.databaseId,
   488→          title: nested.title,
   489→          slug: nested.slug,
   490→          uri: nested.uri,
   491→          content: nested.content || '',
   492→          image: nested.featuredImage?.node?.sourceUrl || undefined,
   493→          policyFirechat: nested.policyFirechat?.policyFirechat || undefined,
   494→        })
   495→      );
   496→
   497→      return {
   498→        id: child.id,
   499→        databaseId: child.databaseId,
   500→        title: child.title,
   501→        slug: child.slug,
   502→        uri: child.uri,
   503→        description: '', // Will use content or description from WP
   504→        content: child.content || '',
   505→        icon: iconMap[child.slug] || 'FileSearch',
   506→        image: child.featuredImage?.node?.sourceUrl || undefined,
   507→        children: nestedChildren.length > 0 ? nestedChildren : undefined,
   508→        policyFirechat: child.policyFirechat?.policyFirechat || undefined,
   509→      };
   510→    });
   511→  } catch {
   512→    return [];
   513→  }

   599→/**
   600→ * Valid team member categories for ACF teamCategory select field
   601→ */
   602→const VALID_TEAM_CATEGORIES = ['executive', 'director', 'advisory-board', 'board-of-trustees'] as const;
   603→
   604→/**
   605→ * Get team members data from WordPress.
   606→ *
   607→ * Strategy: Try ACF first, then fall back to category-based approach.
   608→ *
   609→ * 1. ACF approach (preferred): Fetches posts with ACF "teamMember" field group.
   610→ *    - Requires ACF field group "Team Member" with GraphQL type "TeamMember"
   611→ *    - Uses a "team" category to scope the query
   612→ *    - Each post has ACF fields: teamName, teamRole, teamCategory, teamBio, teamImage
   613→ *
   614→ * 2. Category-based fallback (legacy): Fetches posts from 4 separate categories.
   615→ *    - Post Title = Person's Name
   616→ *    - First Tag = Role/Position
   617→ *    - Post Content/Excerpt = Bio
   618→ *    - Featured Image = Profile picture
   619→ *    - Category = Which group they belong to
   620→ */
   621→export async function getTeamMembers(): Promise<GTEEPTeamMember[]> {
   622→  // --- Try ACF approach first ---
   623→  try {
   624→    const acfResponse = await fetchGraphQL<any>(GET_TEAM_MEMBERS_ACF);
   625→
   626→    if (!acfResponse.errors && acfResponse.data?.posts?.nodes?.length) {
   627→      const posts = acfResponse.data.posts.nodes;
   628→
   629→      // Check if any posts actually have ACF teamMember data
   630→      const hasAcfData = posts.some(
   631→        (p: any) => p.teamMember && (p.teamMember.teamName || p.teamMember.teamRole)
   632→      );
   633→
   634→      if (hasAcfData) {
   635→        const teamMembers: GTEEPTeamMember[] = posts
   636→          .filter((p: any) => p.teamMember?.teamName || p.title)
   637→          .map((p: any) => {
   638→            const acf = p.teamMember || {};
   639→            const category = VALID_TEAM_CATEGORIES.includes(acf.teamCategory)
   640→              ? acf.teamCategory
   641→              : 'director'; // default fallback
   642→
   643→            return {
   644→              id: p.id,
   645→              name: acf.teamName || p.title || '',
   646→              role: acf.teamRole || (p.tags?.nodes?.[0]?.name || category.charAt(0).toUpperCase() + category.slice(1).replace(/-/g, ' ')),
   647→              category,
   648→              bio: acf.teamBio || (p.content
   649→                ? p.content.replace(/<[^>]*>/g, '').trim()
   650→                : p.excerpt?.replace(/<[^>]*>/g, '').trim() || ''),
   651→              image: acf.teamImage?.sourceUrl || p.featuredImage?.node?.sourceUrl || '',
   652→            };
   653→          });
   654→
   655→        return teamMembers;
   656→      }
   657→    }
   658→  } catch {
   659→    // ACF query failed — likely not set up yet. Fall through to category-based.
   660→  }
   661→
   662→  // --- Category-based fallback (original approach) ---
   663→  try {
   664→    const response = await fetchGraphQL<any>(GET_TEAM_MEMBERS);
   665→
   666→    if (response.errors || !response.data) {
   667→      return [];
   668→    }
   669→
   670→    const mapPostToMember = (
   671→      post: any,
   672→      category: GTEEPTeamMember['category']
   673→    ): GTEEPTeamMember => ({
   674→      id: post.id,
   675→      name: post.title || '',
   676→      role: post.tags?.nodes?.[0]?.name || category.charAt(0).toUpperCase() + category.slice(1).replace(/-/g, ' '),
   677→      category,
   678→      bio: post.content
   679→        ? post.content.replace(/<[^>]*>/g, '').trim()
   680→        : post.excerpt?.replace(/<[^>]*>/g, '').trim() || '',
   681→      image: post.featuredImage?.node?.sourceUrl || '',
   682→    });
   683→
   684→    const { executives, directors, advisoryBoard, boardOfTrustees } = response.data;
   685→
   686→    const teamMembers: GTEEPTeamMember[] = [
   687→      ...(executives?.nodes || []).map((p: any) => mapPostToMember(p, 'executive')),
   688→      ...(directors?.nodes || []).map((p: any) => mapPostToMember(p, 'director')),
   689→      ...(advisoryBoard?.nodes || []).map((p: any) => mapPostToMember(p, 'advisory-board')),
   690→      ...(boardOfTrustees?.nodes || []).map((p: any) => mapPostToMember(p, 'board-of-trustees')),
   691→    ];
   692→
   693→    return teamMembers;
   694→  } catch {
   695→    return [];
   696→  }
   697→}

   731→/**
   732→ * Get blog posts data from WordPress.
   733→ */
   734→export async function getBlogPosts(): Promise<import('@/types').GTEEPBlogPost[]> {
   735→  const { posts } = await getPosts(50);
   736→
   737→  // Categories used for team members — these should NOT appear in blog posts
   738→  const teamCategorySlugs = ['executive', 'director', 'advisory-board', 'board-of-trustees', 'team'];
   739→
   740→  // Filter out team member posts and the default "Hello world!" WP post
   741→  const blogPosts = posts.filter((post) => {
   742→    if (post.slug === 'hello-world') return false;
   743→
   744→    const postCategories = (post.categories as any)?.nodes || [];
   745→    const hasTeamCategory = postCategories.some((cat: any) =>
   746→      teamCategorySlugs.includes(cat.slug)
   747→    );
   748→    if (hasTeamCategory) return false;
   749→
   750→    return true;
   751→  });
   752→
   753→  if (blogPosts.length > 0) {
   754→    return blogPosts.map((post): import('@/types').GTEEPBlogPost => ({
   755→      id: post.id,
   756→      title: post.title,
   757→      slug: post.slug,
   758→      excerpt: post.excerpt || '',
   759→      content: post.content || '',
   760→      date: post.date,
   761→      author: (post.author as any)?.node?.name || 'GTEEP Team',
   762→      image: post.featuredImage?.node?.sourceUrl || undefined,
   763→      categories: (post.categories as any)?.nodes?.map((c: any) => c.name) || [],
   764→    }));
   765→  }
   766→
   767→  return [];
   768→}
