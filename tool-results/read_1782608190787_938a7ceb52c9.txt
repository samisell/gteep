    77→// Helper: Extract nodes from a GraphQL response
    78→// -----------------------------------------------------------------------------
    79→
    80→function extractNodes<T>(
    81→  response: { data?: any; errors?: any[] },
    82→  path: string
    83→): T[] {
    84→  if (response.errors || !response.data) {
    85→    return [];
    86→  }
    87→
    88→  const parts = path.split('.');
    89→  let result: any = response.data;
    90→  for (const part of parts) {
    91→    if (result?.[part] === undefined) return [];
    92→    result = result[part];
    93→  }
    94→
    95→  if (result?.nodes && Array.isArray(result.nodes) && result.nodes.length > 0) {
    96→    return result.nodes as T[];
    97→  }
    98→
    99→  return [];
   100→}
   101→
   102→function extractSingleNode<T>(
   103→  response: { data?: any; errors?: any[] },
   104→  key: string
   105→): T | null {
   106→  if (response.errors || !response.data) {
   107→    return null;
   108→  }
   109→
   110→  const node = response.data[key];
   111→  if (!node) return null;
   112→
   113→  return node as T;
   114→}
   115→
   116→function extractPageInfo(
   117→  response: { data?: any; errors?: any[] },
   118→  path: string
   119→): WPPageInfo | null {
   120→  if (response.errors || !response.data) return null;
   121→
   122→  const parts = path.split('.');
   123→  let result: any = response.data;
   124→  for (const part of parts) {
   125→    if (result?.[part] === undefined) return null;
   126→    result = result[part];
   127→  }
   128→
   129→  return result?.pageInfo ?? null;
   130→}

   186→// -----------------------------------------------------------------------------
   187→// About Page
   188→// Fetches the About Us page from WordPress.
   189→//
   190→// The "about summary" shown on the About Us page is sourced from the
   191→// WordPress page editor (page.content) — i.e. what the site admin edits in
   192→// the standard WordPress block editor. This is the canonical source of
   193→// truth so that edits to the About Us page in WordPress are reflected on
   194→// the site immediately.
   195→//
   196→// The ACF `aboutContent.aboutSummary` field is kept as a fallback only,
   197→// for backwards compatibility when the page editor content is empty.
   198→// Vision / Mission / Goal still come from the ACF `aboutContent` group.
   199→// -----------------------------------------------------------------------------
   200→
   201→/**
   202→ * Convert WordPress `page.content` HTML into plain-text paragraphs joined
   203→ * by "\n\n" so the existing rendering logic (which splits on "\n\n") keeps
   204→ * working unchanged. Only <p>...</p> blocks are extracted; empty paragraphs
   205→ * are skipped. Common HTML entities are decoded to their plain-text form.
   206→ */
   207→function htmlContentToParagraphs(html: string | undefined | null): string {
   208→  if (!html) return '';
   209→  const paragraphs: string[] = [];
   210→  const regex = /<p[^>]*>([\s\S]*?)<\/p>/gi;
   211→  let match: RegExpExecArray | null;
   212→  while ((match = regex.exec(html)) !== null) {
   213→    const text = match[1]
   214→      .replace(/<[^>]+>/g, '') // strip inner HTML tags
   215→      .replace(/&nbsp;/g, ' ')
   216→      .replace(/&amp;/g, '&')
   217→      .replace(/&lt;/g, '<')
   218→      .replace(/&gt;/g, '>')
   219→      .replace(/&quot;/g, '"')
   220→      .replace(/&#039;|&apos;|&rsquo;|&#8217;/g, "'")
   221→      .replace(/&lsquo;|&#8216;/g, "'")
   222→      .replace(/&ldquo;|&#8220;/g, '"')
   223→      .replace(/&rdquo;|&#8221;/g, '"')
   224→      .replace(/&ndash;|&#8211;/g, '-')
   225→      .replace(/&mdash;|&#8212;/g, '—')
   226→      .replace(/\s+/g, ' ')
   227→      .trim();
   228→    if (text) paragraphs.push(text);
   229→  }
   230→  return paragraphs.join('\n\n');
   231→}

   312→// -----------------------------------------------------------------------------
   313→// Admin Email (fetches ACF adminemail field from WP)
   314→// The adminemail ACF field group is on the Post type, with a single field:
   315→//   adminEmailForNotification — the email that receives contact/newsletter notifications
   316→// -----------------------------------------------------------------------------
   317→
   318→const DEFAULT_ADMIN_EMAIL = 'info@gteep.gileadtrust.com';
   319→
   320→// Cache admin email for 5 minutes to avoid repeated WP queries
   321→let adminEmailCache: { email: string; timestamp: number } | null = null;
   322→const ADMIN_EMAIL_CACHE_TTL = 5 * 60 * 1000; // 5 minutes
   323→
   324→export async function getAdminEmail(): Promise<string> {
   325→  // Return cached value if still valid
   326→  if (adminEmailCache && Date.now() - adminEmailCache.timestamp < ADMIN_EMAIL_CACHE_TTL) {
   327→    return adminEmailCache.email;
   328→  }
   329→
   330→  try {
   331→    const response = await fetchGraphQL<any>(GET_ADMIN_EMAIL);
   332→
   333→    if (!response.errors && response.data?.posts?.nodes?.length) {
   334→      const email = response.data.posts.nodes[0]?.adminemail?.adminEmailForNotification;
   335→      if (email && /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
   336→        adminEmailCache = { email, timestamp: Date.now() };
   337→        return email;
   338→      }
   339→    }
   340→  } catch {
   341→    // Fall through to default
   342→  }
   343→
   344→  // Fall back to env var or default
   345→  const fallback = process.env.CONTACT_RECEIVER_EMAIL || process.env.SITE_EMAIL || DEFAULT_ADMIN_EMAIL;
   346→  return fallback;
   347→}

   400→ * Fetch the site logo URL from WordPress.

   402→export async function getSiteLogo(): Promise<string | null> {
   403→  try {
   404→    const response = await fetchGraphQL<any>(`
   405→      query GetSiteLogo {
   406→        mediaItems(first: 10, where: { search: "cropped-refined_logo" }) {
   407→          nodes {
   408→            id
   409→            sourceUrl
   410→            altText
   411→            title
   412→          }
   413→        }
   414→      }
   415→    `);
   416→
   417→    if (response.errors || !response.data?.mediaItems?.nodes?.length) {
   418→      const fallbackResponse = await fetchGraphQL<any>(`
   419→        query GetSiteIcon {
   420→          mediaItem(id: "8", idType: DATABASE_ID) {
   421→            sourceUrl
   422→            altText
   423→          }
   424→        }
   425→      `);
   426→
   427→      if (fallbackResponse.errors || !fallbackResponse.data?.mediaItem?.sourceUrl) {
   428→        return null;
   429→      }
   430→
   431→      return fallbackResponse.data.mediaItem.sourceUrl;
   432→    }
   433→
   434→    const logo = response.data.mediaItems.nodes.find(
   435→      (node: any) => node.title?.includes('cropped-refined_logo') || node.sourceUrl?.includes('cropped-refined_logo')
   436→    );
   437→
   438→    return logo?.sourceUrl || response.data.mediaItems.nodes[0]?.sourceUrl || null;
   439→  } catch {
   440→    return null;
   441→  }

   444→export async function getSiteSettings(): Promise<WPSiteSettings> {

   447→  if (response.errors || !response.data) {

   523→    }

   526→    const childNodes = page.children?.nodes || [];
   527→
   528→    if (childNodes.length === 0) {
   529→      return [];
   530→    }

   663→ */
   664→const VALID_TEAM_CATEGORIES = ['executive', 'director', 'advisory-board', 'board-of-trustees'] as const;
   665→
   666→/**
   667→ * Get team members data from WordPress.
   668→ *
   669→ * Strategy: Try ACF first, then fall back to category-based approach.
   670→ *
   671→ * 1. ACF approach (preferred): Fetches posts with ACF "teamMember" field group.
   672→ *    - Requires ACF field group "Team Member" with GraphQL type "TeamMember"
   673→ *    - Uses a "team" category to scope the query
   674→ *    - Each post has ACF fields: teamName, teamRole, teamCategory, teamBio, teamImage
   675→ *
   676→ * 2. Category-based fallback (legacy): Fetches posts from 4 separate categories.
   677→ *    - Post Title = Person's Name
   678→ *    - First Tag = Role/Position
   679→ *    - Post Content/Excerpt = Bio
   680→ *    - Featured Image = Profile picture
   681→ *    - Category = Which group they belong to
   682→ */
   683→export async function getTeamMembers(): Promise<GTEEPTeamMember[]> {
   684→  // --- Try ACF approach first ---
   685→  try {
   686→    const acfResponse = await fetchGraphQL<any>(GET_TEAM_MEMBERS_ACF);
   687→
   688→    if (!acfResponse.errors && acfResponse.data?.posts?.nodes?.length) {
   689→      const posts = acfResponse.data.posts.nodes;
   690→
   691→      // Check if any posts actually have ACF teamMember data
   692→      const hasAcfData = posts.some(
   693→        (p: any) => p.teamMember && (p.teamMember.teamName || p.teamMember.teamRole)
   694→      );
   695→
   696→      if (hasAcfData) {
   697→        const teamMembers: GTEEPTeamMember[] = posts
   698→          .filter((p: any) => p.teamMember?.teamName || p.title)
   699→          .map((p: any) => {
   700→            const acf = p.teamMember || {};
   701→            const category = VALID_TEAM_CATEGORIES.includes(acf.teamCategory)
   702→              ? acf.teamCategory
   703→              : 'director'; // default fallback
   704→
   705→            return {
   706→              id: p.id,
   707→              name: acf.teamName || p.title || '',
   708→              role: acf.teamRole || (p.tags?.nodes?.[0]?.name || category.charAt(0).toUpperCase() + category.slice(1).replace(/-/g, ' ')),
   709→              category,
   710→              bio: acf.teamBio || (p.content
   711→                ? p.content.replace(/<[^>]*>/g, '').trim()
   712→                : p.excerpt?.replace(/<[^>]*>/g, '').trim() || ''),
   713→              image: acf.teamImage?.sourceUrl || p.featuredImage?.node?.sourceUrl || '',
   714→            };
   715→          });
   716→
   717→        return teamMembers;
   718→      }
   719→    }
   720→  } catch {
   721→    // ACF query failed — likely not set up yet. Fall through to category-based.
   722→  }
   723→
   724→  // --- Category-based fallback (original approach) ---
   725→  try {
   726→    const response = await fetchGraphQL<any>(GET_TEAM_MEMBERS);
   727→
   728→    if (response.errors || !response.data) {
   729→      return [];
   730→    }
   731→
   732→    const mapPostToMember = (
   733→      post: any,
   734→      category: GTEEPTeamMember['category']
   735→    ): GTEEPTeamMember => ({
   736→      id: post.id,
   737→      name: post.title || '',
   738→      role: post.tags?.nodes?.[0]?.name || category.charAt(0).toUpperCase() + category.slice(1).replace(/-/g, ' '),
   739→      category,
   740→      bio: post.content
   741→        ? post.content.replace(/<[^>]*>/g, '').trim()
   742→        : post.excerpt?.replace(/<[^>]*>/g, '').trim() || '',
   743→      image: post.featuredImage?.node?.sourceUrl || '',
   744→    });
   745→
   746→    const { executives, directors, advisoryBoard, boardOfTrustees } = response.data;
   747→
   748→    const teamMembers: GTEEPTeamMember[] = [
   749→      ...(executives?.nodes || []).map((p: any) => mapPostToMember(p, 'executive')),
   750→      ...(directors?.nodes || []).map((p: any) => mapPostToMember(p, 'director')),
   751→      ...(advisoryBoard?.nodes || []).map((p: any) => mapPostToMember(p, 'advisory-board')),
   752→      ...(boardOfTrustees?.nodes || []).map((p: any) => mapPostToMember(p, 'board-of-trustees')),
   753→    ];
   754→
   755→    return teamMembers;
   756→  } catch {
   757→    return [];
   758→  }
   759→}
