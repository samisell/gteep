    90→
    91→function AnimatedCounter({ end, suffix = '', duration = 2000 }: { end: number; suffix?: string; duration?: number }) {
    92→  const [count, setCount] = useState(0);
    93→  const ref = useRef<HTMLSpanElement>(null);
    94→  const isInView = useInView(ref, { once: true });
    95→  const hasStarted = useRef(false);
    96→
    97→  useEffect(() => {
    98→    if (!isInView || hasStarted.current) return;
    99→    hasStarted.current = true;
   100→
   101→    const startTime = performance.now();
   102→    const animate = (currentTime: number) => {
   103→      const elapsed = currentTime - startTime;
   104→      const progress = Math.min(elapsed / duration, 1);
   105→      const eased = 1 - Math.pow(1 - progress, 3);
   106→      setCount(Math.floor(eased * end));
   107→      if (progress < 1) {
   108→        requestAnimationFrame(animate);
   109→      }
   110→    };
   111→    requestAnimationFrame(animate);
   112→  }, [isInView, end, duration]);
   113→
   114→  return (
   115→    <span ref={ref} className="tabular-nums">
   116→      {count}{suffix}
   117→    </span>
   118→  );
   119→}
