    57→
    58→function TeamAvatar({ name, size = 'md', imageUrl }: { name: string; size?: 'sm' | 'md' | 'lg'; imageUrl?: string }) {
    59→  const initials = name
    60→    .split(' ')
    61→    .map((n) => n[0])
    62→    .join('')
    63→    .slice(0, 2)
    64→    .toUpperCase();
    65→
    66→  const sizeClasses = {
    67→    sm: 'w-12 h-12 text-sm',
    68→    md: 'w-16 h-16 text-lg',
    69→    lg: 'w-24 h-24 text-2xl',
    70→  };
    71→
    72→  if (imageUrl) {
    73→    return (
    74→      <div className={`${sizeClasses[size]} rounded-full overflow-hidden shadow-lg shrink-0 relative`}>
    75→        <Image
    76→          src={imageUrl}
    77→          alt={name}
    78→          fill
    79→          className="object-cover object-center"
    80→        />
    81→      </div>
    82→    );
    83→  }
    84→
    85→  return (
    86→    <div
    87→      className={`${sizeClasses[size]} rounded-full bg-gradient-to-br from-[#059669] to-[#065f46] flex items-center justify-center text-white font-bold shadow-lg shrink-0`}
    88→      style={{ fontFamily: 'Georgia, "Times New Roman", serif' }}
    89→    >
    90→      {initials}
    91→    </div>
    92→  );
    93→}
