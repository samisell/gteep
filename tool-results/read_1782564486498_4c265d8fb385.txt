    71→function getOutputTypeLabel(type: string): string {
    72→  const labels: Record<string, string> = {
    73→    'concept-note': 'Concept Note',
    74→    'policy-brief': 'Policy Brief',
    75→    'data-stock': 'Data Stock',
    76→    'video': 'Video Gallery',
    77→    'photo': 'Photo Gallery',
    78→    'knowledge-product': 'Knowledge Product',
    79→  };
    80→  return labels[type] || 'Output';

    83→function getOutputTypeBadgeColor(type: string): string {
    84→  switch (type) {
    85→    case 'concept-note': return 'bg-[#f0fdf4] text-[#059669] border-[#065f46]/20';
    86→    case 'policy-brief': return 'bg-[#fef3c7] text-[#d97706] border-[#d97706]/20';

   159→function WpContent({ html, className }: { html: string; className?: string }) {

   164→      dangerouslySetInnerHTML={{ __html: html }}
   165→    />

   167→}
