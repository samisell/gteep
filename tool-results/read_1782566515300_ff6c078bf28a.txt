    74→// -----------------------------------------------------------------------------
    75→// Helper: Extract nodes from a GraphQL response
    76→// -----------------------------------------------------------------------------
    77→
    78→function extractNodes<T>(
    79→  response: { data?: any; errors?: any[] },
    80→  path: string
    81→): T[] {
    82→  if (response.errors || !response.data) {
    83→    return [];
    84→  }
    85→
    86→  const parts = path.split('.');
    87→  let result: any = response.data;
    88→  for (const part of parts) {
    89→    if (result?.[part] === undefined) return [];
    90→    result = result[part];
    91→  }
    92→
    93→  if (result?.nodes && Array.isArray(result.nodes) && result.nodes.length > 0) {
    94→    return result.nodes as T[];
    95→  }
    96→
    97→  return [];
    98→}
    99→
   100→function extractSingleNode<T>(
   101→  response: { data?: any; errors?: any[] },
   102→  key: string
   103→): T | null {
   104→  if (response.errors || !response.data) {
   105→    return null;
   106→  }
   107→
   108→  const node = response.data[key];
   109→  if (!node) return null;
   110→
   111→  return node as T;
   112→}
   113→
   114→function extractPageInfo(
   115→  response: { data?: any; errors?: any[] },
   116→  path: string
   117→): WPPageInfo | null {
   118→  if (response.errors || !response.data) return null;
   119→
   120→  const parts = path.split('.');
   121→  let result: any = response.data;
   122→  for (const part of parts) {
   123→    if (result?.[part] === undefined) return null;
   124→    result = result[part];
   125→  }
   126→
   127→  return result?.pageInfo ?? null;
   128→}

   256→// -----------------------------------------------------------------------------
   257→// Admin Email (fetches ACF adminemail field from WP)
   258→// The adminemail ACF field group is on the Post type, with a single field:
   259→//   adminEmailForNotification — the email that receives contact/newsletter notifications
   260→// -----------------------------------------------------------------------------
   261→
   262→const DEFAULT_ADMIN_EMAIL = 'info@gteep.gileadtrust.com';
   263→
   264→// Cache admin email for 5 minutes to avoid repeated WP queries
   265→let adminEmailCache: { email: string; timestamp: number } | null = null;
   266→const ADMIN_EMAIL_CACHE_TTL = 5 * 60 * 1000; // 5 minutes
   267→
   268→export async function getAdminEmail(): Promise<string> {
   269→  // Return cached value if still valid
   270→  if (adminEmailCache && Date.now() - adminEmailCache.timestamp < ADMIN_EMAIL_CACHE_TTL) {
   271→    return adminEmailCache.email;
   272→  }
   273→
   274→  try {
   275→    const response = await fetchGraphQL<any>(GET_ADMIN_EMAIL);
   276→
   277→    if (!response.errors && response.data?.posts?.nodes?.length) {
   278→      const email = response.data.posts.nodes[0]?.adminemail?.adminEmailForNotification;
   279→      if (email && /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
   280→        adminEmailCache = { email, timestamp: Date.now() };
   281→        return email;
   282→      }
   283→    }
   284→  } catch {
   285→    // Fall through to default
   286→  }
   287→
   288→  // Fall back to env var or default
   289→  const fallback = process.env.CONTACT_RECEIVER_EMAIL || process.env.SITE_EMAIL || DEFAULT_ADMIN_EMAIL;
   290→  return fallback;
   291→}

   343→/**
   344→ * Fetch the site logo URL from WordPress.
   345→ */
   346→export async function getSiteLogo(): Promise<string | null> {
   347→  try {
   348→    const response = await fetchGraphQL<any>(`
   349→      query GetSiteLogo {
   350→        mediaItems(first: 10, where: { search: "cropped-refined_logo" }) {
   351→          nodes {
   352→            id
   353→            sourceUrl
   354→            altText
   355→            title
   356→          }
   357→        }
   358→      }
   359→    `);
   360→
   361→    if (response.errors || !response.data?.mediaItems?.nodes?.length) {
   362→      const fallbackResponse = await fetchGraphQL<any>(`
   363→        query GetSiteIcon {
   364→          mediaItem(id: "8", idType: DATABASE_ID) {
   365→            sourceUrl
   366→            altText
   367→          }
   368→        }
   369→      `);
   370→
   371→      if (fallbackResponse.errors || !fallbackResponse.data?.mediaItem?.sourceUrl) {
   372→        return null;
   373→      }
   374→
   375→      return fallbackResponse.data.mediaItem.sourceUrl;
   376→    }
   377→
   378→    const logo = response.data.mediaItems.nodes.find(
   379→      (node: any) => node.title?.includes('cropped-refined_logo') || node.sourceUrl?.includes('cropped-refined_logo')
   380→    );
   381→
   382→    return logo?.sourceUrl || response.data.mediaItems.nodes[0]?.sourceUrl || null;
   383→  } catch {
   384→    return null;
   385→  }
   386→}

   388→export async function getSiteSettings(): Promise<WPSiteSettings> {
   389→  const response = await fetchGraphQL<WPSiteSettingsData>(GET_SITE_SETTINGS);
   390→
   391→  if (response.errors || !response.data) {
   392→    return defaultSiteSettings;
   393→  }
   394→
   395→  const { generalSettings } = response.data;
   396→
   397→  if (!generalSettings) {
   398→    return defaultSiteSettings;
   399→  }
   400→
   401→  // Try to fetch the site logo from WP
   402→  const logoUrl = await getSiteLogo();
   403→
   404→  // Determine the site description
   405→  const wpDescription = generalSettings.description;
   406→  const isDefaultWpTagline = !wpDescription || wpDescription === 'Just another WordPress site';
   407→  const siteDescription = isDefaultWpTagline ? defaultSiteSettings.siteDescription : wpDescription;
   408→
   409→  return {
   410→    siteTitle: generalSettings.title || defaultSiteSettings.siteTitle,
   411→    siteDescription,
   412→    siteUrl: generalSettings.url || defaultSiteSettings.siteUrl,
   413→    siteLogo: logoUrl

   415→          sourceUrl: logoUrl,
   416→          altText: 'GTEEP Logo',
   417→          mediaItemId: 8,
   418→          width: 512,
   419→          height: 512,

   454→/**
   455→ * Get activities (What We Do) data from WordPress.
   456→ * Fetches the "what-we-do" page and its child pages from WP.
   457→ * Each child page becomes an activity section.
   458→ * Nested children (e.g., Policy Firechat under Policy Engagement) are included.
   459→ * New pages added under "What We Do" in WP will appear automatically.
   460→ */

   462→  try {
   463→    const response = await fetchGraphQL<any>(GET_WHAT_WE_DO_PAGE);
   464→
   465→    if (response.errors || !response.data?.page) {
   466→      return [];
   467→    }
   468→
   469→    const page = response.data.page;
   470→    const childNodes = page.children?.nodes || [];
   471→
   472→    if (childNodes.length === 0) {
   473→      return [];
   474→    }
   475→
   476→    // Map of known icons for specific activities (by slug)
   477→    const iconMap: Record<string, string> = {
   478→      'policy-research': 'FileSearch',
   479→      'policy-engagement': 'Users',
   480→      'citizen-enlightenment': 'Lightbulb',
   481→      'data-speaks': 'BarChart3',
   482→      'youth-mentoring': 'GraduationCap',
   483→      'womens-economic-livelihood': 'Heart',
   484→      'our-publication': 'BookOpen',
   485→    };
   486→
   487→    return childNodes.map((child: any, index: number) => {
   488→      // Parse nested children (e.g., Policy Firechat under Policy Engagement)
   489→      const nestedChildren: import('@/types').GTEEPActivityChild[] = (child.children?.nodes || []).map(
   490→        (nested: any) => ({
   491→          id: nested.id,
   492→          databaseId: nested.databaseId,
   493→          title: nested.title,
   494→          slug: nested.slug,
   495→          uri: nested.uri,
   496→          content: nested.content || '',
   497→          image: nested.featuredImage?.node?.sourceUrl || undefined,
   498→          policyFirechat: nested.policyFirechat?.policyFirechat || undefined,
   499→        })
   500→      );
   501→
   502→      return {
   503→        id: child.id,
   504→        databaseId: child.databaseId,
   505→        title: child.title,
   506→        slug: child.slug,
   507→        uri: child.uri,
   508→        description: '', // Will use content or description from WP
   509→        content: child.content || '',
   510→        icon: iconMap[child.slug] || 'FileSearch',
   511→        image: child.featuredImage?.node?.sourceUrl || undefined,
   512→        children: nestedChildren.length > 0 ? nestedChildren : undefined,
   513→        policyFirechat: child.policyFirechat?.policyFirechat || undefined,
   514→      };
   515→    });
   516→  } catch {
   517→    return [];
   518→  }
   519→}

   605→/**
   606→ * Valid team member categories for ACF teamCategory select field
   607→ */
   608→const VALID_TEAM_CATEGORIES = ['executive', 'director', 'advisory-board', 'board-of-trustees'] as const;
   609→
   610→/**
   611→ * Get team members data from WordPress.
   612→ *
   613→ * Strategy: Try ACF first, then fall back to category-based approach.
   614→ *
   615→ * 1. ACF approach (preferred): Fetches posts with ACF "teamMember" field group.
   616→ *    - Requires ACF field group "Team Member" with GraphQL type "TeamMember"
   617→ *    - Uses a "team" category to scope the query
   618→ *    - Each post has ACF fields: teamName, teamRole, teamCategory, teamBio, teamImage
   619→ *
   620→ * 2. Category-based fallback (legacy): Fetches posts from 4 separate categories.
   621→ *    - Post Title = Person's Name
   622→ *    - First Tag = Role/Position
   623→ *    - Post Content/Excerpt = Bio
   624→ *    - Featured Image = Profile picture
   625→ *    - Category = Which group they belong to
   626→ */
   627→export async function getTeamMembers(): Promise<GTEEPTeamMember[]> {
   628→  // --- Try ACF approach first ---
   629→  try {
   630→    const acfResponse = await fetchGraphQL<any>(GET_TEAM_MEMBERS_ACF);
   631→
   632→    if (!acfResponse.errors && acfResponse.data?.posts?.nodes?.length) {
   633→      const posts = acfResponse.data.posts.nodes;
   634→
   635→      // Check if any posts actually have ACF teamMember data
   636→      const hasAcfData = posts.some(
   637→        (p: any) => p.teamMember && (p.teamMember.teamName || p.teamMember.teamRole)
   638→      );
   639→
   640→      if (hasAcfData) {
   641→        const teamMembers: GTEEPTeamMember[] = posts
   642→          .filter((p: any) => p.teamMember?.teamName || p.title)
   643→          .map((p: any) => {
   644→            const acf = p.teamMember || {};
   645→            const category = VALID_TEAM_CATEGORIES.includes(acf.teamCategory)
   646→              ? acf.teamCategory
   647→              : 'director'; // default fallback
   648→
   649→            return {
   650→              id: p.id,
   651→              name: acf.teamName || p.title || '',
   652→              role: acf.teamRole || (p.tags?.nodes?.[0]?.name || category.charAt(0).toUpperCase() + category.slice(1).replace(/-/g, ' ')),
   653→              category,
   654→              bio: acf.teamBio || (p.content
   655→                ? p.content.replace(/<[^>]*>/g, '').trim()
   656→                : p.excerpt?.replace(/<[^>]*>/g, '').trim() || ''),
   657→              image: acf.teamImage?.sourceUrl || p.featuredImage?.node?.sourceUrl || '',
   658→            };
   659→          });
   660→
   661→        return teamMembers;
   662→      }
   663→    }
   664→  } catch {
   665→    // ACF query failed — likely not set up yet. Fall through to category-based.
   666→  }
   667→
   668→  // --- Category-based fallback (original approach) ---
   669→  try {
   670→    const response = await fetchGraphQL<any>(GET_TEAM_MEMBERS);
   671→
   672→    if (response.errors || !response.data) {
   673→      return [];
   674→    }
   675→
   676→    const mapPostToMember = (
   677→      post: any,
   678→      category: GTEEPTeamMember['category']
   679→    ): GTEEPTeamMember => ({
   680→      id: post.id,
   681→      name: post.title || '',
   682→      role: post.tags?.nodes?.[0]?.name || category.charAt(0).toUpperCase() + category.slice(1).replace(/-/g, ' '),
   683→      category,
   684→      bio: post.content
   685→        ? post.content.replace(/<[^>]*>/g, '').trim()
   686→        : post.excerpt?.replace(/<[^>]*>/g, '').trim() || '',
   687→      image: post.featuredImage?.node?.sourceUrl || '',
   688→    });
   689→
   690→    const { executives, directors, advisoryBoard, boardOfTrustees } = response.data;
   691→
   692→    const teamMembers: GTEEPTeamMember[] = [
   693→      ...(executives?.nodes || []).map((p: any) => mapPostToMember(p, 'executive')),
   694→      ...(directors?.nodes || []).map((p: any) => mapPostToMember(p, 'director')),
   695→      ...(advisoryBoard?.nodes || []).map((p: any) => mapPostToMember(p, 'advisory-board')),
   696→      ...(boardOfTrustees?.nodes || []).map((p: any) => mapPostToMember(p, 'board-of-trustees')),
   697→    ];
   698→
   699→    return teamMembers;
   700→  } catch {
   701→    return [];
   702→  }
   703→}

   740→export async function getBlogPosts(): Promise<import('@/types').GTEEPBlogPost[]> {
   741→  const { posts } = await getPosts(50);
   742→
   743→  // Categories used for team members — these should NOT appear in blog posts
   744→  const teamCategorySlugs = ['executive', 'director', 'advisory-board', 'board-of-trustees', 'team'];
   745→
   746→  // Filter out team member posts and the default "Hello world!" WP post
   747→  const blogPosts = posts.filter((post) => {
   748→    if (post.slug === 'hello-world') return false;
   749→
   750→    const postCategories = (post.categories as any)?.nodes || [];
   751→    const hasTeamCategory = postCategories.some((cat: any) =>
   752→      teamCategorySlugs.includes(cat.slug)
   753→    );
   754→    if (hasTeamCategory) return false;
   755→
   756→    return true;
   757→  });
   758→
   759→  if (blogPosts.length > 0) {
   760→    return blogPosts.map((post): import('@/types').GTEEPBlogPost => ({
   761→      id: post.id,
   762→      title: post.title,
   763→      slug: post.slug,
   764→      excerpt: post.excerpt || '',
   765→      content: post.content || '',
   766→      date: post.date,
   767→      author: (post.author as any)?.node?.name || 'GTEEP Team',
   768→      image: post.featuredImage?.node?.sourceUrl || undefined,
   769→      categories: (post.categories as any)?.nodes?.map((c: any) => c.name) || [],
   770→    }));
   771→  }
   772→
   773→  return [];
   774→}

   914→interface VideoGalleryPageData {

   935→export async function getVideoGallery(): Promise<YouTubeVideo[]> {
   936→  try {
   937→    const response = await fetchGraphQL<VideoGalleryPageData>(GET_VIDEO_GALLERY);
   938→
   939→    if (response.errors || !response.data?.page) {
   940→      return [];
   941→    }
   942→
   943→    const page = response.data.page;
   944→    const videoGallery = page.videoGallery;
   945→    if (!videoGallery) return [];
   946→
   947→    // Collect all YouTube URLs from ACF fields
   948→    const videoUrls: string[] = [];
   949→
   950→    // Single URL fields
   951→    if (videoGallery.firechatEvent && videoGallery.firechatEvent.trim()) {
   952→      videoUrls.push(videoGallery.firechatEvent.trim());
   953→    }
   954→
   955→    // Repeater field support (when user adds 'videos' repeater in ACF):
   956→    // if (videoGallery.videos && Array.isArray(videoGallery.videos)) {
   957→    //   for (const row of videoGallery.videos) {
   958→    //     if (row.videoUrl && row.videoUrl.trim()) {
   959→    //       videoUrls.push(row.videoUrl.trim());
   960→    //     }
   961→    //   }
   962→    // }
   963→
   964→    if (videoUrls.length === 0) return [];
   965→
   966→    // Extract video IDs and enrich with oEmbed data
   967→    const { extractVideoId, fetchVideoByOembed } = await import('@/lib/youtube');
   968→
   969→    const videoPromises = videoUrls.map(async (url) => {
   970→      const videoId = extractVideoId(url);
   971→      if (!videoId) return null;
   972→      return fetchVideoByOembed(videoId);
   973→    });
   974→
   975→    const results = await Promise.all(videoPromises);
   976→
   977→    // Filter out null results and de-duplicate by videoId
   978→    const seen = new Set<string>();
   979→    const videos: YouTubeVideo[] = [];
   980→
   981→    for (const video of results) {
   982→      if (video && !seen.has(video.videoId)) {
   983→        seen.add(video.videoId);
   984→        // Mark as ACF-managed video (not "other video")
   985→        videos.push({
   986→          ...video,
   987→          isOtherVideo: false,
   988→        });
   989→      }
   990→    }
   991→
   992→    return videos;
   993→  } catch (error) {
   994→    console.error('Failed to fetch video gallery from ACF:', error);
   995→    return [];
   996→  }

  1051→function getFileExtension(url: string): string {
  1052→  try {
  1053→    const pathname = new URL(url).pathname;
  1054→    const ext = pathname.split('.').pop()?.toLowerCase() || '';
  1055→    return ext;
  1056→  } catch {
  1057→    return '';
  1058→  }
  1059→}

  1061→function getFileTypeIcon(ext: string): string {
  1062→  switch (ext) {
  1063→    case 'pptx': case 'ppt': return 'presentation';
  1064→    case 'docx': case 'doc': return 'document';
  1065→    case 'pdf': return 'pdf';
  1066→    case 'xlsx': case 'xls': return 'spreadsheet';
  1067→    default: return 'file';
  1068→  }

  1091→      return [];

  1096→    const downloadables = post.ourOutputDownloadables;
  1097→    if (!downloadables) return [];
  1098→
  1099→    const outputs: GTEEPOutput[] = [];
  1100→    let counter = 0;

  1110→      outputs.push({
  1111→        id: `download-${counter++}`,
  1112→        title: metadata.title,
