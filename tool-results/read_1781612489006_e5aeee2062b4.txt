    81→// Helper: Animated Counter Component
    82→// =============================================================================
    83→
    84→function AnimatedCounter({ end, suffix = '', duration = 2000 }: { end: number; suffix?: string; duration?: number }) {
    85→  const [count, setCount] = useState(0);
    86→  const ref = useRef<HTMLSpanElement>(null);
    87→  const isInView = useInView(ref, { once: true });
    88→  const hasStarted = useRef(false);
    89→
    90→  useEffect(() => {
    91→    if (!isInView || hasStarted.current) return;
    92→    hasStarted.current = true;
    93→
    94→    const startTime = performance.now();
    95→    const animate = (currentTime: number) => {
    96→      const elapsed = currentTime - startTime;
    97→      const progress = Math.min(elapsed / duration, 1);
    98→      const eased = 1 - Math.pow(1 - progress, 3);
    99→      setCount(Math.floor(eased * end));
   100→      if (progress < 1) {
   101→        requestAnimationFrame(animate);
   102→      }
   103→    };
   104→    requestAnimationFrame(animate);
   105→  }, [isInView, end, duration]);
   106→
   107→  return (
   108→    <span ref={ref} className="tabular-nums">
   109→      {count}{suffix}
   110→    </span>
   111→  );
   112→}

   219→        <Image
   220→          src={imageUrl}
   221→          alt={name}
   222→          fill
   223→          className="object-cover object-center"
   224→          sizes="96px"
   225→        />
   226→      </div>
   227→    );
   228→  }
   229→
   230→  return (
   231→    <div
   232→      className={`${sizeClasses[size]} rounded-full bg-gradient-to-br from-[#059669] to-[#065f46] flex items-center justify-center text-white font-bold shadow-lg shrink-0`}
