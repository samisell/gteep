    82→function getOutputTypeLabel(type: string): string {
    83→  const labels: Record<string, string> = {
    84→    'concept-note': 'Concept Note',
    85→    'policy-brief': 'Policy Brief',
    86→    'data-stock': 'Data Stock',
    87→    'video': 'Video Gallery',
    88→    'photo': 'Photo Gallery',
    89→    'knowledge-product': 'Knowledge Product',
    90→  };
    91→  return labels[type] || 'Output';
    92→}
    93→
    94→function getOutputTypeBadgeColor(type: string): string {
    95→  switch (type) {
    96→    case 'concept-note': return 'bg-[#f0fdf4] text-[#059669] border-[#065f46]/20';
    97→    case 'policy-brief': return 'bg-[#fef3c7] text-[#d97706] border-[#d97706]/20';
    98→    case 'data-stock': return 'bg-[#eff6ff] text-[#2563eb] border-[#2563eb]/20';
    99→    case 'video': return 'bg-[#faf5ff] text-[#7c3aed] border-[#7c3aed]/20';
   100→    case 'photo': return 'bg-[#fff1f2] text-[#e11d48] border-[#e11d48]/20';
   101→    case 'knowledge-product': return 'bg-[#f0fdfa] text-[#0d9488] border-[#0d9488]/20';
   102→    default: return 'bg-[#f1f5f9] text-[#64748b] border-[#64748b]/20';

   106→function getOutputTypeBgGradient(type: string): string {

   114→    default: return 'from-[#065f46] to-[#0f172a]';

   170→function WpContent({ html, className }: { html: string; className?: string }) {
   171→  if (!html || html.trim() === '') return null;

   173→    <div
   174→      className={`prose prose-slate max-w-none text-[#475569] leading-relaxed [&_h1]:text-2xl [&_h1]:font-bold [&_h1]:text-[#0f172a] [&_h1]:mb-4 [&_h2]:text-xl [&_h2]:font-bold [&_h2]:text-[#0f172a] [&_h2]:mb-3 [&_h3]:text-lg [&_h3]:font-semibold [&_h3]:text-[#0f172a] [&_h3]:mb-2 [&_p]:mb-4 [&_ul]:mb-4 [&_ul]:list-disc [&_ul]:pl-6 [&_ol]:mb-4 [&_ol]:list-decimal [&_ol]:pl-6 [&_li]:mb-1 [&_a]:text-[#059669] [&_a]:underline [&_a:hover]:text-[#047857] [&_strong]:text-[#0f172a] [&_blockquote]:border-l-4 [&_blockquote]:border-[#065f46] [&_blockquote]:pl-4 [&_blockquote]:italic [&_blockquote]:text-[#64748b] ${className || ''}`}
   175→      dangerouslySetInnerHTML={{ __html: html }}
   176→    />
   177→  );
   178→}
