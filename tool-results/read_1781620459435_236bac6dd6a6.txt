   193→   193→
   194→   194→  const validate = () => {
   195→   195→    const newErrors: Record<string, string> = {};
   196→   196→    if (!formData.name.trim()) newErrors.name = 'Full name is required';
   197→   197→    if (!formData.email.trim()) newErrors.email = 'Email is required';
   198→   198→    else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(formData.email)) newErrors.email = 'Please enter a valid email';

   200→   200→    if (!formData.subject.trim()) newErrors.subject = 'Subject is required';
   201→   201→    if (!formData.message.trim()) newErrors.message = 'Message is required';
   202→   202→    else if (formData.message.trim().length < 10) newErrors.message = 'Message must be at least 10 characters';
   203→   203→    if (!consent) newErrors.consent = 'You must consent to data processing';
   204→   204→    setErrors(newErrors);
   205→   205→    return Object.keys(newErrors).length === 0;
   206→   206→  };
   207→   207→
   208→   208→  const handleSubmit = async (e: React.FormEvent) => {
   209→   209→    e.preventDefault();

   212→   212→    setIsSubmitting(true);

   214→   214→    try {

   216→   216→        method: 'POST',
   217→   217→        headers: { 'Content-Type': 'application/json' },
   218→   218→        body: JSON.stringify({
   219→   219→          name: formData.name,
   220→   220→          email: formData.email,
   221→   221→          subject: `[${formData.subjectCategory}] ${formData.subject}`,
   222→   222→          message: formData.message,
   223→   223→          organization: formData.organization,
   224→   224→          phone: formData.phone,
   225→   225→          consent,
   226→   226→        }),
   227→   227→      });
   228→   228→
   229→   229→      const data = await response.json();
   230→   230→
   231→   231→      if (response.ok && data.success) {
   232→   232→        setIsSuccess(true);
   233→   233→        toast.success('Message sent!', {
   234→   234→          description: 'Thank you for reaching out. We will get back to you shortly.',
   235→   235→        });
   236→   236→      } else {
   237→   237→        const errorMsg = data.error || 'Something went wrong. Please try again.';
   238→   238→        toast.error('Submission failed', { description: errorMsg });
   239→   239→      }
   240→   240→    } catch {
   241→   241→      toast.error('Network error', {
   242→   242→        description: 'Could not connect to the server. Please check your internet connection and try again.',
   243→   243→      });
   244→   244→    } finally {
   245→   245→      setIsSubmitting(false);
   246→   246→    }
   247→   247→  };

   249→   249→  const handleChange = (field: string, value: string) => {
   250→   250→    setFormData((prev) => ({ ...prev, [field]: value }));

   252→   252→      setErrors((prev) => {
   253→   253→        const next = { ...prev };
   254→   254→        delete next[field];
   255→   255→        return next;
   256→   256→      });
   257→   257→    }
   258→   258→  };

   275→   275→          <motion.div

   277→   277→            initial="hidden"

   285→   285→                  <CardContent className="p-6">
   286→   286→                    <div
   287→   287→                      className={`w-12 h-12 rounded-xl flex items-center justify-center mb-4 transition-transform duration-300 group-hover:scale-110 ${
   288→   288→                        card.color === 'emerald'

   290→   290→                          : 'bg-[#d97706]/10 text-[#d97706]'

   314→   314→                      >

   316→   316→                        <span className="text-xs">→</span>
   317→   317→                      </a>
   318→   318→                    )}
