    89→
    90→function AnimatedCounter({ end, suffix = '', duration = 2000 }: { end: number; suffix?: string; duration?: number }) {
    91→  const [count, setCount] = useState(0);
    92→  const ref = useRef<HTMLSpanElement>(null);
    93→  const isInView = useInView(ref, { once: true });
    94→  const hasStarted = useRef(false);
    95→
    96→  useEffect(() => {
    97→    if (!isInView || hasStarted.current) return;
    98→    hasStarted.current = true;
    99→
   100→    const startTime = performance.now();
   101→    const animate = (currentTime: number) => {
   102→      const elapsed = currentTime - startTime;
   103→      const progress = Math.min(elapsed / duration, 1);
   104→      const eased = 1 - Math.pow(1 - progress, 3);
   105→      setCount(Math.floor(eased * end));
   106→      if (progress < 1) {
   107→        requestAnimationFrame(animate);
   108→      }
   109→    };
   110→    requestAnimationFrame(animate);
   111→  }, [isInView, end, duration]);
   112→
   113→  return (
   114→    <span ref={ref} className="tabular-nums">
   115→      {count}{suffix}
   116→    </span>
   117→  );
   118→}

   230→          sizes="96px"
   231→        />
