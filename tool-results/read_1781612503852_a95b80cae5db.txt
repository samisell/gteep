   137→// Download Form Dialog Component
   138→// =============================================================================
   139→
   140→function DownloadFormDialog({ fileUrl, fileName }: { fileUrl: string; fileName: string }) {
   141→  const [email, setEmail] = useState('');
   142→  const [name, setName] = useState('');
   143→  const [isSubmitting, setIsSubmitting] = useState(false);
   144→  const [isSubmitted, setIsSubmitted] = useState(false);
   145→  const [isOpen, setIsOpen] = useState(false);
   146→
   147→  const handleSubmit = async (e: React.FormEvent) => {
   148→    e.preventDefault();
   149→    if (!email.trim()) return;
   150→
   151→    setIsSubmitting(true);
   152→    try {
   153→      const res = await fetch('/api/download/request', {
   154→        method: 'POST',
   155→        headers: { 'Content-Type': 'application/json' },
   156→        body: JSON.stringify({ email, name, fileUrl, fileName }),
   157→      });
   158→      if (res.ok) {
   159→        setIsSubmitted(true);
   160→        // Also start the download directly
   161→        window.open(fileUrl, '_blank');
   162→      }
   163→    } catch {
   164→      // Still allow download even if API fails
   165→      window.open(fileUrl, '_blank');
   166→      setIsSubmitted(true);
   167→    } finally {
   168→      setIsSubmitting(false);
   169→    }
   170→  };
   171→
   172→  const handleOpenChange = (open: boolean) => {
   173→    setIsOpen(open);
   174→    if (!open) {
   175→      // Reset after a short delay so the user doesn't see the form reset
   176→      setTimeout(() => {
   177→        setIsSubmitted(false);
   178→        setEmail('');
   179→        setName('');
   180→      }, 300);
   181→    }
   182→  };
   183→
   184→  return (
   185→    <Dialog open={isOpen} onOpenChange={handleOpenChange}>
   186→      <DialogTrigger asChild>
   187→        <Button
   188→          size="sm"
   189→          className="bg-emerald-700 hover:bg-emerald-800 text-white rounded-lg"
   190→        >
   191→          <Download className="w-4 h-4 mr-2" />
   192→          Download Slides
   193→        </Button>
   194→      </DialogTrigger>
   195→      <DialogContent className="sm:max-w-md">
   196→        <DialogHeader>
   197→          <DialogTitle className="flex items-center gap-2">
   198→            <FileText className="w-5 h-5 text-emerald-600" />
   199→            Download Presentation
   200→          </DialogTitle>
   201→          <DialogDescription>
   202→            {isSubmitted
   203→              ? 'Your download has started!'
   204→              : 'Enter your details to download the presentation slides.'}
   205→          </DialogDescription>
   206→        </DialogHeader>
   207→        {isSubmitted ? (
   208→          <motion.div
   209→            initial={{ opacity: 0, scale: 0.9 }}
   210→            animate={{ opacity: 1, scale: 1 }}
   211→            className="py-8 text-center"
   212→          >
   213→            <CheckCircle2 className="w-16 h-16 text-emerald-500 mx-auto mb-4" />
   214→            <h3 className="text-lg font-semibold text-foreground mb-2">
   215→              Download Started!
   216→            </h3>
   217→            <p className="text-sm text-muted-foreground mb-4">
   218→              Thank you, {name || 'friend'}! Your download should begin automatically.
   219→              We&apos;ve also sent a confirmation to <strong>{email}</strong>.
   220→            </p>
   221→            <Button
   222→              variant="outline"
   223→              onClick={() => handleOpenChange(false)}
   224→              className="rounded-lg"
   225→            >
   226→              Close
   227→            </Button>
   228→          </motion.div>
   229→        ) : (
   230→          <form onSubmit={handleSubmit} className="space-y-4 pt-2">
   231→            <div className="space-y-2">
   232→              <label htmlFor="download-name" className="text-sm font-medium">
   233→                Full Name
   234→              </label>
   235→              <Input
   236→                id="download-name"
   237→                type="text"
   238→                placeholder="Your full name"
   239→                value={name}
   240→                onChange={(e) => setName(e.target.value)}
   241→                className="rounded-lg"
   242→              />
   243→            </div>
   244→            <div className="space-y-2">
   245→              <label htmlFor="download-email" className="text-sm font-medium">
   246→                Email Address <span className="text-red-500">*</span>
   247→              </label>
   248→              <Input
   249→                id="download-email"
   250→                type="email"
   251→                placeholder="your@email.com"
   252→                value={email}
   253→                onChange={(e) => setEmail(e.target.value)}
   254→                required
   255→                className="rounded-lg"
   256→              />
   257→            </div>
   258→            <p className="text-xs text-muted-foreground">
   259→              By downloading, you agree to receive occasional policy updates from GTEEP.
   260→              You can unsubscribe at any time.
   261→            </p>
   262→            <Button
   263→              type="submit"
   264→              disabled={isSubmitting || !email.trim()}
   265→              className="w-full bg-emerald-700 hover:bg-emerald-800 text-white rounded-lg"
   266→            >
   267→              {isSubmitting ? (
   268→                <>
   269→                  <Loader2 className="w-4 h-4 mr-2 animate-spin" />
   270→                  Processing...
   271→                </>
   272→              ) : (
   273→                <>
   274→                  <Download className="w-4 h-4 mr-2" />
   275→                  Download Now
   276→                </>
   277→              )}
   278→            </Button>
   279→          </form>
   280→        )}
   281→      </DialogContent>
   282→    </Dialog>
   283→  );
   284→}
