    64→function TeamAvatar({ name, size = 'md', imageUrl }: { name: string; size?: 'sm' | 'md' | 'lg'; imageUrl?: string }) {
    65→  const initials = name
    66→    .split(' ')
    67→    .map((n) => n[0])
    68→    .join('')
    69→    .slice(0, 2)
    70→    .toUpperCase();
    71→
    72→  const sizeClasses = {
    73→    sm: 'w-12 h-12 text-sm',
    74→    md: 'w-16 h-16 text-lg',
    75→    lg: 'w-24 h-24 text-2xl',
    76→  };
    77→
    78→  if (imageUrl) {
    79→    return (
    80→      <div className={`${sizeClasses[size]} rounded-full overflow-hidden shadow-lg shrink-0 relative`}>
    81→        <Image
    82→          src={imageUrl}
    83→          alt={name}
    84→          fill
    85→          className="object-cover object-center"
    86→        />
    87→      </div>
    88→    );
    89→  }
    90→
    91→  return (
    92→    <div
    93→      className={`${sizeClasses[size]} rounded-full bg-gradient-to-br from-[#059669] to-[#065f46] flex items-center justify-center text-white font-bold shadow-lg shrink-0`}
    94→      style={{ fontFamily: 'Georgia, "Times New Roman", serif' }}
    95→    >
    96→      {initials}
    97→    </div>
    98→  );
    99→}
