   188→    if (!formData.subject.trim()) newErrors.subject = 'Subject is required';
   189→    if (!formData.message.trim()) newErrors.message = 'Message is required';
   190→    else if (formData.message.trim().length < 10) newErrors.message = 'Message must be at least 10 characters';
   191→    if (!consent) newErrors.consent = 'You must consent to data processing';
   192→    setErrors(newErrors);
   193→    return Object.keys(newErrors).length === 0;
   194→  };

   197→    e.preventDefault();

   202→    try {
   203→      const response = await fetch('/api/contact', {
   204→        method: 'POST',
   205→        headers: { 'Content-Type': 'application/json' },
   206→        body: JSON.stringify({
   207→          name: formData.name,
   208→          email: formData.email,
   209→          subject: `[${formData.subjectCategory}] ${formData.subject}`,
   210→          message: formData.message,
   211→          organization: formData.organization,
   212→          phone: formData.phone,
   213→          consent,
   214→        }),
