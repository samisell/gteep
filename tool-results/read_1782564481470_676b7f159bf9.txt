    66→function TeamAvatar({ name, size = 'md', imageUrl }: { name: string; size?: 'sm' | 'md' | 'lg'; imageUrl?: string }) {
    67→  const initials = name
    68→    .split(' ')
    69→    .map((n) => n[0])
    70→    .join('')
    71→    .slice(0, 2)
    72→    .toUpperCase();
    73→
    74→  const sizeClasses = {
    75→    sm: 'w-12 h-12 text-sm',
    76→    md: 'w-16 h-16 text-lg',
    77→    lg: 'w-24 h-24 text-2xl',
    78→  };
    79→
    80→  if (imageUrl) {
    81→    return (
    82→      <div className={`${sizeClasses[size]} rounded-full overflow-hidden shadow-lg shrink-0 relative`}>
    83→        <Image
    84→          src={imageUrl}
    85→          alt={name}
    86→          fill
    87→          className="object-cover object-center"
    88→        />
    89→      </div>
    90→    );
    91→  }
    92→
    93→  return (
    94→    <div
    95→      className={`${sizeClasses[size]} rounded-full bg-gradient-to-br from-[#059669] to-[#065f46] flex items-center justify-center text-white font-bold shadow-lg shrink-0`}
    96→      style={{ fontFamily: 'Georgia, "Times New Roman", serif' }}
    97→    >
    98→      {initials}
    99→    </div>
   100→  );
   101→}
