    95→
    96→function AnimatedCounter({ end, suffix = '', duration = 2000 }: { end: number; suffix?: string; duration?: number }) {
    97→  const [count, setCount] = useState(0);
    98→  const ref = useRef<HTMLSpanElement>(null);
    99→  const isInView = useInView(ref, { once: true });
   100→  const hasStarted = useRef(false);
   101→
   102→  useEffect(() => {
   103→    if (!isInView || hasStarted.current) return;
   104→    hasStarted.current = true;
   105→
   106→    const startTime = performance.now();
   107→    const animate = (currentTime: number) => {
   108→      const elapsed = currentTime - startTime;
   109→      const progress = Math.min(elapsed / duration, 1);
   110→      const eased = 1 - Math.pow(1 - progress, 3);
   111→      setCount(Math.floor(eased * end));
   112→      if (progress < 1) {
   113→        requestAnimationFrame(animate);
   114→      }
   115→    };
   116→    requestAnimationFrame(animate);
   117→  }, [isInView, end, duration]);
   118→
   119→  return (
   120→    <span ref={ref} className="tabular-nums">
   121→      {count}{suffix}
   122→    </span>
   123→  );
   124→}

   256→        />
