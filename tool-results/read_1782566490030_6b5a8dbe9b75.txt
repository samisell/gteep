    82→function getOutputTypeLabel(type: string): string {
    83→  const labels: Record<string, string> = {
    84→    'concept-note': 'Concept Note',
    85→    'policy-brief': 'Policy Brief',
    86→    'data-stock': 'Data Stock',
    87→    'video': 'Video Gallery',
    88→    'photo': 'Photo Gallery',
    89→    'knowledge-product': 'Knowledge Product',
    90→  };
    91→  return labels[type] || 'Output';

    94→function getOutputTypeBadgeColor(type: string): string {
    95→  switch (type) {
    96→    case 'concept-note': return 'bg-[#f0fdf4] text-[#059669] border-[#065f46]/20';
    97→    case 'policy-brief': return 'bg-[#fef3c7] text-[#d97706] border-[#d97706]/20';
