   139→   139→
   140→   140→function DownloadFormDialog({ fileUrl, fileName }: { fileUrl: string; fileName: string }) {
   141→   141→  const [email, setEmail] = useState('');
   142→   142→  const [name, setName] = useState('');
   143→   143→  const [isSubmitting, setIsSubmitting] = useState(false);
   144→   144→  const [isSubmitted, setIsSubmitted] = useState(false);
   145→   145→  const [isOpen, setIsOpen] = useState(false);
   146→   146→
   147→   147→  const handleSubmit = async (e: React.FormEvent) => {
   148→   148→    e.preventDefault();
   149→   149→    if (!email.trim()) return;
   150→   150→
   151→   151→    setIsSubmitting(true);
   152→   152→    try {
   153→   153→      const res = await fetch('/api/download/request', {
   154→   154→        method: 'POST',
   155→   155→        headers: { 'Content-Type': 'application/json' },
   156→   156→        body: JSON.stringify({ email, name, fileUrl, fileName }),
   157→   157→      });
   158→   158→      if (res.ok) {
   159→   159→        setIsSubmitted(true);
   160→   160→        // Also start the download directly
   161→   161→        window.open(fileUrl, '_blank');
   162→   162→      }
   163→   163→    } catch {
   164→   164→      // Still allow download even if API fails
   165→   165→      window.open(fileUrl, '_blank');
   166→   166→      setIsSubmitted(true);
   167→   167→    } finally {
   168→   168→      setIsSubmitting(false);
   169→   169→    }
   170→   170→  };
   171→   171→
   172→   172→  const handleOpenChange = (open: boolean) => {
   173→   173→    setIsOpen(open);
   174→   174→    if (!open) {
   175→   175→      // Reset after a short delay so the user doesn't see the form reset
   176→   176→      setTimeout(() => {
   177→   177→        setIsSubmitted(false);
   178→   178→        setEmail('');
   179→   179→        setName('');
   180→   180→      }, 300);
   181→   181→    }
   182→   182→  };
   183→   183→
   184→   184→  return (
   185→   185→    <Dialog open={isOpen} onOpenChange={handleOpenChange}>
   186→   186→      <DialogTrigger asChild>
   187→   187→        <Button
   188→   188→          size="sm"
   189→   189→          className="bg-emerald-700 hover:bg-emerald-800 text-white rounded-lg"
   190→   190→        >
   191→   191→          <Download className="w-4 h-4 mr-2" />
   192→   192→          Download Slides
   193→   193→        </Button>
   194→   194→      </DialogTrigger>
   195→   195→      <DialogContent className="sm:max-w-md">
   196→   196→        <DialogHeader>
   197→   197→          <DialogTitle className="flex items-center gap-2">
   198→   198→            <FileText className="w-5 h-5 text-emerald-600" />
   199→   199→            Download Presentation
   200→   200→          </DialogTitle>
   201→   201→          <DialogDescription>
   202→   202→            {isSubmitted
   203→   203→              ? 'Your download has started!'
   204→   204→              : 'Enter your details to download the presentation slides.'}
   205→   205→          </DialogDescription>
   206→   206→        </DialogHeader>
   207→   207→        {isSubmitted ? (
   208→   208→          <motion.div
   209→   209→            initial={{ opacity: 0, scale: 0.9 }}
   210→   210→            animate={{ opacity: 1, scale: 1 }}
   211→   211→            className="py-8 text-center"
   212→   212→          >
   213→   213→            <CheckCircle2 className="w-16 h-16 text-emerald-500 mx-auto mb-4" />
   214→   214→            <h3 className="text-lg font-semibold text-foreground mb-2">
   215→   215→              Download Started!
   216→   216→            </h3>
   217→   217→            <p className="text-sm text-muted-foreground mb-4">
   218→   218→              Thank you, {name || 'friend'}! Your download should begin automatically.
   219→   219→              We&apos;ve also sent a confirmation to <strong>{email}</strong>.
   220→   220→            </p>
   221→   221→            <Button
   222→   222→              variant="outline"
   223→   223→              onClick={() => handleOpenChange(false)}
   224→   224→              className="rounded-lg"
   225→   225→            >
   226→   226→              Close
   227→   227→            </Button>
   228→   228→          </motion.div>
   229→   229→        ) : (
   230→   230→          <form onSubmit={handleSubmit} className="space-y-4 pt-2">
   231→   231→            <div className="space-y-2">
   232→   232→              <label htmlFor="download-name" className="text-sm font-medium">
   233→   233→                Full Name
   234→   234→              </label>
   235→   235→              <Input
   236→   236→                id="download-name"
   237→   237→                type="text"
   238→   238→                placeholder="Your full name"
   239→   239→                value={name}
   240→   240→                onChange={(e) => setName(e.target.value)}
   241→   241→                className="rounded-lg"
   242→   242→              />
   243→   243→            </div>
   244→   244→            <div className="space-y-2">
   245→   245→              <label htmlFor="download-email" className="text-sm font-medium">
   246→   246→                Email Address <span className="text-red-500">*</span>
   247→   247→              </label>
   248→   248→              <Input
   249→   249→                id="download-email"
   250→   250→                type="email"
   251→   251→                placeholder="your@email.com"
   252→   252→                value={email}
   253→   253→                onChange={(e) => setEmail(e.target.value)}
   254→   254→                required
   255→   255→                className="rounded-lg"
   256→   256→              />
   257→   257→            </div>
   258→   258→            <p className="text-xs text-muted-foreground">
   259→   259→              By downloading, you agree to receive occasional policy updates from GTEEP.
   260→   260→              You can unsubscribe at any time.
   261→   261→            </p>
   262→   262→            <Button
   263→   263→              type="submit"
   264→   264→              disabled={isSubmitting || !email.trim()}
   265→   265→              className="w-full bg-emerald-700 hover:bg-emerald-800 text-white rounded-lg"
   266→   266→            >
   267→   267→              {isSubmitting ? (
   268→   268→                <>
   269→   269→                  <Loader2 className="w-4 h-4 mr-2 animate-spin" />
   270→   270→                  Processing...
   271→   271→                </>
   272→   272→              ) : (
   273→   273→                <>
   274→   274→                  <Download className="w-4 h-4 mr-2" />
   275→   275→                  Download Now
   276→   276→                </>
   277→   277→              )}
   278→   278→            </Button>
   279→   279→          </form>
   280→   280→        )}
   281→   281→      </DialogContent>
   282→   282→    </Dialog>
   283→   283→  );
   284→   284→}
