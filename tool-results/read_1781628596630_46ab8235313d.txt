    73→    73→// Helper: Extract nodes from a GraphQL response

    75→    75→
    76→    76→function extractNodes<T>(
    77→    77→  response: { data?: any; errors?: any[] },
    78→    78→  path: string
    79→    79→): T[] {
    80→    80→  if (response.errors || !response.data) {
    81→    81→    return [];
    82→    82→  }
    83→    83→
    84→    84→  const parts = path.split('.');
    85→    85→  let result: any = response.data;
    86→    86→  for (const part of parts) {
    87→    87→    if (result?.[part] === undefined) return [];
    88→    88→    result = result[part];
    89→    89→  }
    90→    90→
    91→    91→  if (result?.nodes && Array.isArray(result.nodes) && result.nodes.length > 0) {
    92→    92→    return result.nodes as T[];
    93→    93→  }
    94→    94→
    95→    95→  return [];
    96→    96→}
    97→    97→
    98→    98→function extractSingleNode<T>(
    99→    99→  response: { data?: any; errors?: any[] },
   100→   100→  key: string
   101→   101→): T | null {
   102→   102→  if (response.errors || !response.data) {
   103→   103→    return null;
   104→   104→  }
   105→   105→
   106→   106→  const node = response.data[key];
   107→   107→  if (!node) return null;
   108→   108→
   109→   109→  return node as T;
   110→   110→}

   254→   254→// -----------------------------------------------------------------------------
   255→   255→// Admin Email (fetches ACF adminemail field from WP)
   256→   256→// The adminemail ACF field group is on the Post type, with a single field:
   257→   257→//   adminEmailForNotification — the email that receives contact/newsletter notifications
   258→   258→// -----------------------------------------------------------------------------
   259→   259→
   260→   260→const DEFAULT_ADMIN_EMAIL = 'info@gteep.gileadtrust.com';
   261→   261→
   262→   262→// Cache admin email for 5 minutes to avoid repeated WP queries
   263→   263→let adminEmailCache: { email: string; timestamp: number } | null = null;
   264→   264→const ADMIN_EMAIL_CACHE_TTL = 5 * 60 * 1000; // 5 minutes
   265→   265→
   266→   266→export async function getAdminEmail(): Promise<string> {
   267→   267→  // Return cached value if still valid
   268→   268→  if (adminEmailCache && Date.now() - adminEmailCache.timestamp < ADMIN_EMAIL_CACHE_TTL) {
   269→   269→    return adminEmailCache.email;
   270→   270→  }
   271→   271→
   272→   272→  try {
   273→   273→    const response = await fetchGraphQL<any>(GET_ADMIN_EMAIL);
   274→   274→
   275→   275→    if (!response.errors && response.data?.posts?.nodes?.length) {
   276→   276→      const email = response.data.posts.nodes[0]?.adminemail?.adminEmailForNotification;
   277→   277→      if (email && /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
   278→   278→        adminEmailCache = { email, timestamp: Date.now() };
   279→   279→        return email;
   280→   280→      }
   281→   281→    }
   282→   282→  } catch {
   283→   283→    // Fall through to default
   284→   284→  }
   285→   285→
   286→   286→  // Fall back to env var or default
   287→   287→  const fallback = process.env.CONTACT_RECEIVER_EMAIL || process.env.SITE_EMAIL || DEFAULT_ADMIN_EMAIL;
   288→   288→  return fallback;
   289→   289→}

   349→   349→          nodes {

   359→   359→    if (response.errors || !response.data?.mediaItems?.nodes?.length) {
   360→   360→      const fallbackResponse = await fetchGraphQL<any>(`
   361→   361→        query GetSiteIcon {

   364→   364→            altText

   370→   370→        return null;

   372→   372→
   373→   373→      return fallbackResponse.data.mediaItem.sourceUrl;

   377→   377→      (node: any) => node.title?.includes('cropped-refined_logo') || node.sourceUrl?.includes('cropped-refined_logo')

   380→   380→    return logo?.sourceUrl || response.data.mediaItems.nodes[0]?.sourceUrl || null;

   454→   454→ * Fetches the "what-we-do" page and its child pages from WP.
   455→   455→ * Each child page becomes an activity section.
   456→   456→ * Nested children (e.g., Policy Firechat under Policy Engagement) are included.

   459→   459→export async function getActivities(): Promise<import('@/types').GTEEPActivity[]> {

   463→   463→    if (response.errors || !response.data?.page) {
   464→   464→      return [];
   465→   465→    }
   466→   466→
   467→   467→    const page = response.data.page;
   468→   468→    const childNodes = page.children?.nodes || [];
   469→   469→
   470→   470→    if (childNodes.length === 0) {
   471→   471→      return [];
   472→   472→    }
   473→   473→
   474→   474→    // Map of known icons for specific activities (by slug)

   476→   476→      'policy-research': 'FileSearch',
   477→   477→      'policy-engagement': 'Users',
   478→   478→      'citizen-enlightenment': 'Lightbulb',
   479→   479→      'data-speaks': 'BarChart3',
   480→   480→      'youth-mentoring': 'GraduationCap',
   481→   481→      'womens-economic-livelihood': 'Heart',
   482→   482→    };
   483→   483→
   484→   484→    return childNodes.map((child: any, index: number) => {
   485→   485→      // Parse nested children (e.g., Policy Firechat under Policy Engagement)
   486→   486→      const nestedChildren: import('@/types').GTEEPActivityChild[] = (child.children?.nodes || []).map(
   487→   487→        (nested: any) => ({
   488→   488→          id: nested.id,
   489→   489→          databaseId: nested.databaseId,
   490→   490→          title: nested.title,
   491→   491→          slug: nested.slug,
   492→   492→          uri: nested.uri,

   494→   494→          image: nested.featuredImage?.node?.sourceUrl || undefined,
   495→   495→          policyFirechat: nested.policyFirechat?.policyFirechat || undefined,
   496→   496→        })
   497→   497→      );
   498→   498→
   499→   499→      return {
   500→   500→        id: child.id,
   501→   501→        databaseId: child.databaseId,
   502→   502→        title: child.title,
   503→   503→        slug: child.slug,
   504→   504→        uri: child.uri,
   505→   505→        description: '', // Will use content or description from WP
   506→   506→        content: child.content || '',
   507→   507→        icon: iconMap[child.slug] || 'FileSearch',
   508→   508→        image: child.featuredImage?.node?.sourceUrl || undefined,
   509→   509→        children: nestedChildren.length > 0 ? nestedChildren : undefined,
   510→   510→        policyFirechat: child.policyFirechat?.policyFirechat || undefined,

   514→   514→    return [];

   613→   613→ *    - Uses a "team" category to scope the query
   614→   614→ *    - Each post has ACF fields: teamName, teamRole, teamCategory, teamBio, teamImage
   615→   615→ *
   616→   616→ * 2. Category-based fallback (legacy): Fetches posts from 4 separate categories.
   617→   617→ *    - Post Title = Person's Name
   618→   618→ *    - First Tag = Role/Position
   619→   619→ *    - Post Content/Excerpt = Bio
   620→   620→ *    - Featured Image = Profile picture
   621→   621→ *    - Category = Which group they belong to

   623→   623→export async function getTeamMembers(): Promise<GTEEPTeamMember[]> {

   629→   629→      const posts = acfResponse.data.posts.nodes;
   630→   630→

   632→   632→      const hasAcfData = posts.some(
   633→   633→        (p: any) => p.teamMember && (p.teamMember.teamName || p.teamMember.teamRole)
   634→   634→      );

   645→   645→            return {
   646→   646→              id: p.id,
   647→   647→              name: acf.teamName || p.title || '',
   648→   648→              role: acf.teamRole || (p.tags?.nodes?.[0]?.name || category.charAt(0).toUpperCase() + category.slice(1).replace(/-/g, ' ')),
   649→   649→              category,
   650→   650→              bio: acf.teamBio || (p.content
   651→   651→                ? p.content.replace(/<[^>]*>/g, '').trim()

   664→   664→  // --- Category-based fallback (original approach) ---

   679→   679→      category,
   680→   680→      bio: post.content
   681→   681→        ? post.content.replace(/<[^>]*>/g, '').trim()
   682→   682→        : post.excerpt?.replace(/<[^>]*>/g, '').trim() || '',
   683→   683→      image: post.featuredImage?.node?.sourceUrl || '',
   684→   684→    });
   685→   685→
   686→   686→    const { executives, directors, advisoryBoard, boardOfTrustees } = response.data;
   687→   687→
   688→   688→    const teamMembers: GTEEPTeamMember[] = [
   689→   689→      ...(executives?.nodes || []).map((p: any) => mapPostToMember(p, 'executive')),
   690→   690→      ...(directors?.nodes || []).map((p: any) => mapPostToMember(p, 'director')),
   691→   691→      ...(advisoryBoard?.nodes || []).map((p: any) => mapPostToMember(p, 'advisory-board')),
   692→   692→      ...(boardOfTrustees?.nodes || []).map((p: any) => mapPostToMember(p, 'board-of-trustees')),
   693→   693→    ];
